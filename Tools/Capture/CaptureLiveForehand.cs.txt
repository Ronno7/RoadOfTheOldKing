// Fresh Tutorial Play Mode. Stop afterward to discard temporary equipment; no save session.
int direction = 2; // 0 east, 1 north, 2 south, 3 west
Vector2 aim=new[]{Vector2.right,Vector2.up,Vector2.down,Vector2.left}[direction];
string facing=new[]{"East","North","South","West"}[direction];
if(!Application.isPlaying || UnityEngine.SceneManagement.SceneManager.GetActiveScene().name!="Tutorial" ||
    UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Progression.CheckpointSession>()!=null)
    throw new Exception("Capture in Tutorial Play Mode without a save session.");
var player=GameObject.Find("Player").GetComponent<TheLostShrine.Player.PlayerCombatController>();
var weapon=GameObject.Find("Hatchet Pickup").GetComponent<TheLostShrine.Weapons.HatchetWeapon>();
var animation=player.GetComponent<TheLostShrine.Player.PlayerLocomotionAnimation>();
var movement=player.GetComponent<TheLostShrine.Player.PlayerMovement>();
var motor=player.GetComponent<Rigidbody2D>();
var presenter=player.GetComponent<TheLostShrine.Player.RegisteredPlayerAnimation>();
var view=weapon.GetComponent<TheLostShrine.Weapons.HatchetView>();
var cam=Camera.main;
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var oldSimulation=Physics2D.simulationMode;var oldTarget=cam.targetTexture;var oldActive=RenderTexture.active;
var oldPosition=motor.position;var oldCamera=cam.transform.position;var oldVelocity=motor.linearVelocity;
var oldFacing=movement.FacingDirection;var oldInterpolation=motor.interpolation;
var target=new RenderTexture(1920,1080,24,RenderTextureFormat.ARGB32);
var pixels=new Texture2D(1920,1080,TextureFormat.RGB24,false);
var folder=System.IO.Path.GetFullPath(System.IO.Path.Combine(Application.dataPath,"../../tmp/forehand-proof/native-"+facing.ToLowerInvariant()));
System.IO.Directory.CreateDirectory(folder);
var rows=new System.Collections.Generic.List<string>();GameObject dummy=null;
try {
    Physics2D.simulationMode=SimulationMode2D.Script;motor.interpolation=RigidbodyInterpolation2D.None;
    player.GetType().GetMethod("ProcessInput",flags).Invoke(player,new object[]{new TheLostShrine.Input.CombatInputFrame{Active=true}});
    if(player.Weapon==null && !player.TryEquip(weapon))throw new Exception("Cannot equip the Tutorial axe.");
    weapon.CancelAction();player.GetComponent<TheLostShrine.Player.PlayerStamina>().Restore();
    player.transform.position=new Vector3(45,54,0);motor.position=player.transform.position;motor.linearVelocity=Vector2.zero;
    movement.GetType().GetProperty("FacingDirection").SetValue(movement,aim,null);Physics2D.SyncTransforms();
    cam.transform.position=player.transform.position+new Vector3(1.3f,1.6f,-10);cam.targetTexture=target;
    for(int i=0;i<150;i++) {
        if(i==20 || i==90)weapon.TryLightChop(aim);
        if(i==70) {
            dummy=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Combat/PracticeDummy.prefab"));
            dummy.transform.position=player.transform.position+(Vector3)aim*1.5f;
            dummy.GetComponent<Rigidbody2D>().position=dummy.transform.position;Physics2D.SyncTransforms();
        }
        // Render the accepted pose before the next step, including the first load exposure.
        animation.GetType().GetMethod("Present",flags).Invoke(animation,new object[]{.02f});
        view.GetType().GetMethod("LateUpdate",flags).Invoke(view,null);
        if(dummy!=null) {
            // The capture blocks regular Update; show authoritative health rather than prefab placeholder text.
            var hp=dummy.GetComponent<TheLostShrine.Combat.Damageable>();
            dummy.GetComponentInChildren<TextMesh>().text="DUMMY\n"+hp.Health+"/"+hp.MaxHealth;
            var feedback=dummy.GetComponent<TheLostShrine.Prototype.PracticeTarget>();
            var fill=(Transform)feedback.GetType().GetField("healthFill",flags).GetValue(feedback);
            fill.localScale=new Vector3((float)hp.Health/hp.MaxHealth,1,1);
        }
        cam.Render();RenderTexture.active=target;pixels.ReadPixels(new Rect(0,0,1920,1080),0,0);pixels.Apply();
        System.IO.File.WriteAllBytes(System.IO.Path.Combine(folder,i.ToString("D3")+".png"),pixels.EncodeToPNG());
        rows.Add(i+","+presenter.DisplayedAction+","+presenter.DisplayedCel+","+weapon.IsImpactPaused+","+(dummy==null?-1:dummy.GetComponent<TheLostShrine.Combat.Damageable>().Health));
        weapon.GetType().GetMethod("Simulate",flags).Invoke(weapon,new object[]{.02f});
    }
    System.IO.File.WriteAllLines(System.IO.Path.Combine(folder,"timeline.csv"),rows.ToArray());
    return "150 native Tutorial frames: original carry, missed forehand, recovery, actual target hit and contact pause.";
}
finally {
    if(dummy!=null)UnityEngine.Object.DestroyImmediate(dummy);
    weapon.CancelAction();cam.targetTexture=oldTarget;RenderTexture.active=oldActive;cam.transform.position=oldCamera;
    motor.position=oldPosition;player.transform.position=oldPosition;motor.linearVelocity=oldVelocity;motor.interpolation=oldInterpolation;
    movement.GetType().GetProperty("FacingDirection").SetValue(movement,oldFacing,null);Physics2D.simulationMode=oldSimulation;
    target.Release();UnityEngine.Object.DestroyImmediate(target);UnityEngine.Object.DestroyImmediate(pixels);
}
