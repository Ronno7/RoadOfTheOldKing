// Use a fresh PrototypeLoop session with an isolated CheckpointSession saveKey.
// Tests intentionally save progress and leave a completed route at the second fire.
var results=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,name)=>results.Add((ok?"PASS ":"FAIL ")+name);
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var session=TheLostShrine.Progression.CheckpointSession.Instance;
string key=(string)session.GetType().GetField("saveKey",flags).GetValue(session);
if(!key.StartsWith("TheLostShrine.Verification."))throw new System.Exception("Use an isolated verification save slot.");
var player=GameObject.Find("Player").GetComponent<TheLostShrine.Player.PlayerCombatController>();
var health=player.GetComponent<TheLostShrine.Player.PlayerHealth>();var body=player.GetComponent<Rigidbody2D>();
var interaction=player.GetComponent<TheLostShrine.Player.PlayerBonfireInteraction>();
var weapon=UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Weapons.HatchetWeapon>();
var guide=UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Prototype.PrototypeLoopGuide>();
var enemy=UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Combat.SimpleMeleeEnemy>();
var puzzle=UnityEngine.Object.FindFirstObjectByType<TheLostShrine.World.ThrowRecallPuzzle>();
var camp=System.Array.Find(session.Fires,f=>f.Id=="prototype/sentinel-camp");var exit=System.Array.Find(session.Fires,f=>f.Id=="prototype/beyond-gate");
var anchor=GameObject.Find("Throw Anchor").GetComponent<TheLostShrine.World.HatchetPuzzleTarget>();var receiver=GameObject.Find("Recall Switch").GetComponent<TheLostShrine.World.HatchetPuzzleTarget>();
var guideUpdate=guide.GetType().GetMethod("Update",flags);var sim=weapon.GetType().GetMethod("Simulate",flags);
System.Action<float> tick=duration=>{for(int i=0;i<Mathf.CeilToInt(duration/.02f);i++)sim.Invoke(weapon,new object[]{.02f});};
System.Action update=()=>guideUpdate.Invoke(guide,null);
System.Action<Vector2> move=position=>{body.position=position;body.linearVelocity=Vector2.zero;player.transform.position=position;Physics2D.SyncTransforms();};
var store=new TheLostShrine.Progression.PlayerPrefsProgressStore(key);
var oldSimulation=Physics2D.simulationMode;
try{
Physics2D.simulationMode=SimulationMode2D.Script;
check(session.Fires.Length==2&&camp!=null&&exit!=null,"Both uniquely identified bonfires are registered");
check(!session.HasCheckpoint&&!camp.IsDiscovered&&!exit.IsDiscovered,"A fresh save has no checkpoint or travel destinations");
check(!session.Rest(camp)&&health.Health.Health==100,"A distant fire cannot be used to heal or save");
check(!session.Travel(exit,camp),"Travel to an undiscovered fire is rejected");
// Establish the already-verified first section's end state.
player.TryEquip(weapon);player.UnlockRecall();
var hit=new TheLostShrine.Combat.CombatHit(player.gameObject,TheLostShrine.Combat.AttackKind.ChargedCleave,1000,Vector2.up,0,0,true);
foreach(var prop in UnityEngine.Object.FindObjectsByType<TheLostShrine.Combat.Breakable>(FindObjectsSortMode.None))prop.ReceiveHit(hit);
enemy.GetComponent<TheLostShrine.Combat.Damageable>().ReceiveHit(hit);
guide.GetType().GetField("<Step>k__BackingField",flags).SetValue(guide,TheLostShrine.Prototype.PrototypeStep.Enemy);update();
check(guide.Step==TheLostShrine.Prototype.PrototypeStep.Bonfire,"Enemy defeat opens the new campfire section");
health.Health.ReceiveHit(new TheLostShrine.Combat.CombatHit(enemy.gameObject,TheLostShrine.Combat.AttackKind.EnemyMelee,40,Vector2.down,1,.2f));
check(health.Health.Health==60,"Player can arrive at camp with damage");
move(camp.SpawnPosition);weapon.TryThrow(Vector2.right);tick(.04f);
check(interaction.Open(camp),"Opening the nearby fire performs a rest");update();
check(health.Health.Health==100&&!health.GetComponent<TheLostShrine.Combat.HitReaction>().IsStaggered,"Rest fully heals and clears stagger");
check(weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Rest safely retrieves a thrown hatchet");
check(enemy.GetComponent<TheLostShrine.Combat.Damageable>().Health==100&&enemy.State==TheLostShrine.Combat.MeleeEnemyState.Idle&&enemy.GetComponent<Collider2D>().enabled,"Rest revives the defeated enemy at full health");
check(Vector2.Distance(enemy.transform.position,new Vector2(-15.5f,11.5f))<.01f,"Rest restores enemy home position");
check(camp.IsDiscovered&&session.Progress.checkpointId==camp.Id&&guide.Step==TheLostShrine.Prototype.PrototypeStep.Puzzle,"First fire becomes the checkpoint and opens puzzle access");
check(interaction.IsOpen&&!player.GetComponent<TheLostShrine.Input.PlayerMovementInput>().enabled&&!player.GetComponent<TheLostShrine.Input.PlayerCombatInput>().enabled,"Fire menu disables movement and combat input");
var saved=store.Load();
check(saved.hasHatchet&&saved.recallUnlocked&&saved.checkpointId==camp.Id&&saved.discoveredFires.Count==1,"Save round trip preserves equipment, Recall, checkpoint and discovery");
check(saved.completedIds.FindAll(id=>id.StartsWith("prototype/obstacle/")).Count==3,"All cleared route obstacles are saved as permanent progress");
check(!session.Travel(exit,camp),"Resting at first fire does not unlock travel to second fire");
interaction.Close();check(player.GetComponent<TheLostShrine.Input.PlayerMovementInput>().enabled&&player.GetComponent<TheLostShrine.Input.PlayerCombatInput>().enabled,"Leaving the menu restores player input");
check(!puzzle.Door.IsOpen&&!puzzle.IsSolved,"Puzzle door starts closed");
check(!receiver.ReceiveHit(new TheLostShrine.Combat.CombatHit(player.gameObject,TheLostShrine.Combat.AttackKind.Recall,2,Vector2.right,0,0))&&!puzzle.IsSolved,"Recall target rejects activation before the throw target is armed");
check(!anchor.ReceiveHit(new TheLostShrine.Combat.CombatHit(player.gameObject,TheLostShrine.Combat.AttackKind.LightChop,1,Vector2.right,0,0))&&!puzzle.IsArmed,"Melee cannot arm the throw target");
move(new Vector2(6.5f,12.5f));weapon.TryThrow(Vector2.right);tick(.7f);
check(puzzle.IsArmed&&!puzzle.IsSolved&&weapon.State==TheLostShrine.Weapons.HatchetState.Stuck,"Actual outbound throw arms the gold target and sticks");
check(!receiver.ReceiveHit(new TheLostShrine.Combat.CombatHit(player.gameObject,TheLostShrine.Combat.AttackKind.Throw,2,Vector2.right,0,0))&&!puzzle.IsSolved,"Outbound hits cannot activate the Recall target");
weapon.TryRecall();tick(.8f);
check(!puzzle.IsSolved&&weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Recalling from the throw pad misses the switch; repositioning matters");
move(camp.SpawnPosition);session.Rest(camp);
check(!puzzle.IsArmed&&!puzzle.Door.IsOpen,"Rest resets an unfinished puzzle without opening its door");
move(new Vector2(6.5f,12.5f));weapon.TryThrow(Vector2.right);tick(.7f);move(new Vector2(6.5f,8.5f));weapon.TryRecall();tick(.8f);update();
check(puzzle.IsSolved&&puzzle.Door.IsOpen&&weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Actual throw, reposition and Recall opens the next door");
check(guide.Step==TheLostShrine.Prototype.PrototypeStep.ExitBonfire,"Solved puzzle advances the instruction to the far fire");
saved=store.Load();check(saved.Has("prototype/recall-door")&&saved.checkpointId==camp.Id,"Puzzle completion saves immediately without changing the respawn fire");
move(camp.SpawnPosition);session.Rest(camp);update();
check(puzzle.IsSolved&&puzzle.Door.IsOpen&&guide.Step>=TheLostShrine.Prototype.PrototypeStep.ExitBonfire,"Rest never relocks a completed puzzle or regresses route progress");
move(exit.SpawnPosition);check(interaction.Open(exit),"Far fire can be discovered after the door opens");update();
check(session.Progress.discoveredFires.Count==2&&session.Progress.checkpointId==exit.Id&&guide.Step==TheLostShrine.Prototype.PrototypeStep.Complete,"Second fire completes the route and becomes the respawn point");
check(session.Travel(camp,exit),"Travel between two discovered fires succeeds");interaction.Close();
check(Vector2.Distance(body.position,camp.SpawnPosition)<.01f&&session.Progress.checkpointId==camp.Id&&health.Health.Health==100,"Travel moves to a safe spawn, heals, and updates checkpoint");
check(puzzle.Door.IsOpen&&player.CanRecall,"Travel preserves the solved door and Recall");
check(!session.Travel(exit,exit),"Travel cannot be initiated remotely from another fire");
check(session.Travel(exit,camp),"Return travel to the far fire succeeds");
saved=store.Load();check(saved.checkpointId==exit.Id&&saved.Has("prototype/recall-door")&&saved.discoveredFires.Count==2,"Latest save includes the chosen destination and permanent puzzle state");
// Corrupt/versioned-save handling is checked in a separate disposable slot.
string disposable=key+".invalid";PlayerPrefs.SetString(disposable,"{\"version\":999}");bool rejected=false;
try{new TheLostShrine.Progression.PlayerPrefsProgressStore(disposable).Load();}catch(System.Exception){rejected=true;}
check(rejected,"Unsupported save versions are rejected instead of silently misread");
new TheLostShrine.Progression.PlayerPrefsProgressStore(disposable).Clear();check(!PlayerPrefs.HasKey(disposable),"Clearing a save affects only its own slot");
}finally{Physics2D.simulationMode=oldSimulation;}
return string.Join("\n",results.ToArray());
