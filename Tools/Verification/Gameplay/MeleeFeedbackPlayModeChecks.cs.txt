// Empty unsaved Play Mode scene, no GameSession. Uses real player/weapon assets.
// Checks full-sector feedback independently of terrain-blocked actual damage.
if (!Application.isPlaying || UnityEngine.Object.FindFirstObjectByType<RoadOfTheOldKing.Progression.GameSession>() != null)
    throw new Exception("Use an empty Play Mode scene without a checkpoint session.");
var results = new System.Collections.Generic.List<string>();
System.Action<bool,string> check = (ok,label) => results.Add((ok ? "PASS " : "FAIL ") + label);
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic;
var created = new System.Collections.Generic.List<GameObject>();
System.Func<string,GameObject> spawn = path => { var go = UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>(path)); created.Add(go); return go; };
var previousSimulation = Physics2D.simulationMode;
var upgrade = ScriptableObject.CreateInstance<RoadOfTheOldKing.Weapons.AxeUpgrade>();
RoadOfTheOldKing.Weapons.AxeSettings isolated = null;
try {
    Physics2D.simulationMode = SimulationMode2D.Script;
    var root = spawn("Assets/Prefabs/Player/Player.prefab");
    var player = root.GetComponent<RoadOfTheOldKing.Player.PlayerCombatController>();
    var weapon = spawn("Assets/Prefabs/Weapons/Axe.prefab").GetComponent<RoadOfTheOldKing.Weapons.AxeWeapon>();
    var feedback = weapon.GetComponent<RoadOfTheOldKing.Weapons.AxeMeleeFeedback>();
    var surface = (MeshRenderer)feedback.GetType().GetField("surface",flags).GetValue(feedback);
    var mesh = (Mesh)feedback.GetType().GetField("mesh",flags).GetValue(feedback);
    var stamina = root.GetComponent<RoadOfTheOldKing.Player.PlayerStamina>();
    // Isolate window/geometry tests from hit pause with a private settings copy; the shared asset is untouched.
    isolated = UnityEngine.Object.Instantiate(weapon.Settings); isolated.lightHitPause = 0f;
    weapon.GetType().GetField("settings",flags).SetValue(weapon,isolated);
    System.Action<float> tick = dt => weapon.GetType().GetMethod("Simulate",flags).Invoke(weapon,new object[]{dt});
    System.Action render = () => feedback.GetType().GetMethod("LateUpdate",flags).Invoke(feedback,null);
    System.Action reset = () => { weapon.CancelAction(); stamina.Restore(); };
    var target = new GameObject("Melee target"); created.Add(target);
    target.AddComponent<CircleCollider2D>().radius = .025f;
    var health = target.AddComponent<RoadOfTheOldKing.Combat.Damageable>();
    System.Action<Vector2> place = position => { target.transform.position = position; health.RestoreHealth(); Physics2D.SyncTransforms(); };
    check(player.TryEquip(weapon),"Standard player equips feedback-enabled weapon");
    render(); check(!surface.enabled,"Idle has no attack footprint");
    float start = weapon.LightDuration * weapon.Settings.lightWindupFraction;
    float end = weapon.LightDuration * weapon.Settings.lightSwingEndFraction;
    place(Vector2.right); weapon.TryLightChop(Vector2.right); tick(start * .5f); render();
    check(weapon.LightPhase == RoadOfTheOldKing.Weapons.MeleePhase.Windup && surface.enabled && health.Health == 100,"Windup outlines intent without damage");
    tick(start * .5f + .001f); render();
    check(weapon.LightPhase == RoadOfTheOldKing.Weapons.MeleePhase.Active && surface.enabled && health.Health == 90,"Active footprint starts with actual damage");
    tick(.001f); check(health.Health == 90,"Footprint does not cause repeat hits");
    weapon.PauseOnImpact(.04f); float progress = weapon.LightSwingProgress;
    tick(.02f); render(); check(weapon.LightSwingProgress == progress && surface.enabled,"Hit pause holds the active swish clock");
    tick(.2f); render(); check(!surface.enabled,"No damaging-looking tail after the window");
    health.RestoreHealth(); tick(.005f); check(health.Health == 100,"Recovery cannot hit a fresh target");

    for (int d = 0; d < 8; d++) {
        float angle = d * Mathf.PI / 4f; Vector2 aim = new Vector2(Mathf.Cos(angle),Mathf.Sin(angle));
        for (int sample = 0; sample < 4; sample++) {
            reset(); float reach = weapon.Settings.lightRadius;
            float offset = sample < 2 ? 0 : (weapon.LightArc * .5f + (sample == 2 ? -6f : 6f)) * Mathf.Deg2Rad;
            float distance = sample == 0 ? reach - .15f : sample == 1 ? reach + .15f : reach * .7f;
            place(new Vector2(Mathf.Cos(angle + offset),Mathf.Sin(angle + offset)) * distance);
            weapon.TryLightChop(aim); tick(start + .002f); render();
            check((health.Health < 100) == (sample == 0 || sample == 2),"Direction " + d + " actual hit agrees with range/angle sample " + sample);
        }
        bool inside = true;
        foreach(var v in mesh.vertices) if(v.magnitude > weapon.LightReach + .001f || (v.sqrMagnitude > .001f && Vector2.Angle(aim,v) > weapon.LightArc*.5f+.01f)) inside = false;
        check(inside,"Direction " + d + " all visual vertices stay inside damage sector");
        weapon.SetAim(-aim); render();
        check(weapon.AttackDirection == aim,"Direction " + d + " committed sector ignores later cursor movement");
    }

    reset(); place(new Vector2(1.4f,0));
    var wall = new GameObject("Melee wall"); created.Add(wall); wall.transform.position = new Vector3(.7f,0,0);
    var wallCollider = wall.AddComponent<BoxCollider2D>(); wallCollider.size = new Vector2(.1f,6); Physics2D.SyncTransforms();
    weapon.TryLightChop(Vector2.right); tick(start+.002f); render();
    check(health.Health==100,"Solid wall still blocks actual damage");
    var unobstructedVertices=mesh.vertices;
    check(surface.enabled && mesh.vertices.Max(v=>v.x)>weapon.LightReach-.002f,"Footprint retains full reach through a wall");
    wall.transform.position=Vector3.zero;Physics2D.SyncTransforms();render();
    check(surface.enabled && mesh.vertices.SequenceEqual(unobstructedVertices),"Collider overlapping the origin cannot collapse the effect");
    wall.transform.position=new Vector3(.7f,0,0);
    wallCollider.isTrigger=true; reset(); Physics2D.SyncTransforms(); weapon.TryLightChop(Vector2.right); tick(start+.002f); render();
    check(health.Health<100,"Triggers do not block melee");
    wallCollider.isTrigger=false; wall.AddComponent<RoadOfTheOldKing.Combat.Damageable>(); reset();health.RestoreHealth();Physics2D.SyncTransforms();weapon.TryLightChop(Vector2.right);tick(start+.002f);
    check(health.Health<100,"Damage receivers do not occlude targets behind them");
    wall.SetActive(false);target.SetActive(false);

    reset(); upgrade.addedLightArc=40f; upgrade.lightSpeedMultiplier=1.2f; weapon.ApplyUpgrades(new[]{upgrade});
    weapon.TryLightChop(Vector2.right); tick(weapon.LightDuration*weapon.Settings.lightWindupFraction+.001f); render();
    check(Mathf.Abs(weapon.LightArc-180f)<.001f && Mathf.Abs(weapon.LightDuration-.25f)<.001f && weapon.LightPhase==RoadOfTheOldKing.Weapons.MeleePhase.Active,"Wider/faster upgrade changes shared geometry and timing");
    check(Vector2.Angle(Vector2.right,mesh.vertices[1])>70f,"Mesh expands to the upgraded arc");
    tick(.3f); stamina.Restore();weapon.TryLightChop(Vector2.right);tick(.1f);render();
    check(weapon.ComboIndex==1 && surface.enabled && weapon.LightArc < 180f,"Backhand stays narrower after widening");
    tick(.3f);stamina.Restore();weapon.TryLightChop(Vector2.right);render();
    check(weapon.ComboIndex==2 && Mathf.Abs(weapon.LightReach-weapon.Settings.lightRadius*1.3f)<.001f,"Finisher expands reach from gameplay stats");
    tick(weapon.LightDuration*.42f);render();
    float farthest=0;foreach(var v in mesh.vertices) farthest=Mathf.Max(farthest,v.x);
    check(Mathf.Abs(farthest-weapon.LightReach)<.001f,"Finisher contour reaches its actual hit boundary");
    Time.timeScale=0;render();check(!surface.enabled,"Pause hides the footprint");Time.timeScale=1;render();check(surface.enabled,"Resume restores current action phase");
    feedback.enabled=false;check(!surface.enabled,"Disabling feedback clears its renderer");feedback.enabled=true;render();
    weapon.CancelAction();render();check(!surface.enabled,"Cancellation clears footprint immediately on presentation update");
    stamina.TrySpend(stamina.Current);check(!weapon.TryLightChop(Vector2.right),"Exhausted click cannot start a swing");render();check(!surface.enabled,"Rejected click has no swipe");
    stamina.Restore();weapon.TryLightChop(Vector2.right);tick(.6f);render();check(!surface.enabled,"Long physics step cannot leave stale active feedback");
    // Combo contract and thrust lane, including all octants and terrain protection.
    weapon.ApplyUpgrades(new RoadOfTheOldKing.Weapons.AxeUpgrade[0]);
    System.Action<Vector2> thrust = aim => {
        reset(); target.SetActive(false);
        weapon.TryLightChop(aim); tick(weapon.LightDuration+.001f);
        stamina.Restore(); weapon.TryLightChop(aim); tick(weapon.LightDuration+.001f);
        stamina.Restore(); weapon.TryLightChop(aim);
    };
    reset();weapon.TryLightChop(Vector2.right);
    check(weapon.LightArc==160f && weapon.LightDamage==10,"Opening sweep is broad at baseline damage");
    float openerWindup=weapon.LightDuration*weapon.LightWindupFraction;
    tick(weapon.LightDuration+.001f);stamina.Restore();weapon.TryLightChop(Vector2.right);
    check(weapon.LightArc==120f && weapon.LightDamage==10,"Reverse sweep is narrower at equal damage");
    tick(weapon.LightDuration+.001f);stamina.Restore();weapon.TryLightChop(Vector2.right);
    check(weapon.IsLightThrust && weapon.LightDamage==25 && weapon.LightDuration*weapon.LightWindupFraction>openerWindup,"Thrust is stronger with the longest tell");
    tick(weapon.LightDuration*weapon.LightSwingEndFraction);
    var view=weapon.GetComponent<RoadOfTheOldKing.Weapons.AxeView>();
    view.GetType().GetMethod("LateUpdate",flags).Invoke(view,null);
    var blade=(SpriteRenderer)view.GetType().GetField("blade",flags).GetValue(view);
    check(Mathf.Abs(blade.bounds.max.x-root.transform.position.x-weapon.LightReach)<.01f,"Thrust sprite tip reaches the lane end without overshooting");
    upgrade.finisherDamageMultiplier=1.4f;upgrade.addedLightArc=40f;weapon.ApplyUpgrades(new[]{upgrade});
    check(weapon.LightDamage==35 && weapon.LightArc==28f,"Finisher bonus strengthens the poke without widening it");
    weapon.ApplyUpgrades(new RoadOfTheOldKing.Weapons.AxeUpgrade[0]);
    for(int d=0;d<8;d++) {
        Vector2 aim=new Vector2(Mathf.Cos(d*Mathf.PI/4),Mathf.Sin(d*Mathf.PI/4));
        Vector2 side=new Vector2(-aim.y,aim.x);
        for(int sample=0;sample<4;sample++) {
            thrust(aim);target.SetActive(true);
            float depth=sample==1 ? weapon.LightReach+.12f : sample==3 ? -.12f : weapon.LightReach*.75f;
            float lateral=sample==2 ? weapon.LightLaneWidth*.5f+.12f : 0f;
            place(aim*depth+side*lateral);
            tick(weapon.LightDuration*weapon.LightWindupFraction+.001f);render();
            check((health.Health==75)==(sample==0),"Thrust octant "+d+" range/side/rear sample "+sample);
        }
        bool inside=true;
        foreach(var v in mesh.vertices) {
            float x=Vector2.Dot(v,aim),y=Vector2.Dot(v,side);
            if(x<-.001f||x>weapon.LightReach+.001f||Mathf.Abs(y)>weapon.LightLaneWidth*.5f+.001f)inside=false;
        }
        check(inside,"Thrust octant "+d+" lane matches damage bounds");
    }
    thrust(Vector2.right);target.SetActive(true);place(Vector2.right*2);
    wall.SetActive(true);wall.transform.position=Vector2.right;
    UnityEngine.Object.DestroyImmediate(wall.GetComponent<RoadOfTheOldKing.Combat.Damageable>());
    Physics2D.SyncTransforms();tick(weapon.LightDuration*.45f);
    check(health.Health==100,"Thrust cannot damage through a wall");wall.SetActive(false);
    thrust(Vector2.right);target.SetActive(true);place(Vector2.right*2);
    tick(weapon.LightDuration*.45f);tick(.01f);
    check(health.Health==75,"Thrust damages once per attack");target.SetActive(false);
    tick(weapon.LightDuration);stamina.Restore();weapon.TryLightChop(Vector2.right);
    check(weapon.ComboIndex==0,"Completed finisher wraps to opener");
    tick(weapon.LightDuration+weapon.Settings.comboWindow+.1f);tick(weapon.Settings.comboWindow+.1f);
    stamina.Restore();weapon.TryLightChop(Vector2.right);
    check(weapon.ComboIndex==0,"Expired combo restarts at opener");
    thrust(Vector2.right);tick(weapon.LightDuration*.2f);weapon.CancelLightCombo();tick(.5f);
    check(weapon.State==RoadOfTheOldKing.Weapons.AxeState.Held&&weapon.ConsumeActionDisplacement()==Vector2.zero,"Interrupted tell cannot lunge later");
    // Run the actual motor/physics so the lunge cannot bypass solid collision.
    var motor=root.GetComponent<RoadOfTheOldKing.Player.PlayerMovement>();
    var body=root.GetComponent<Rigidbody2D>();body.interpolation=RigidbodyInterpolation2D.None;
    var input=root.GetComponent<RoadOfTheOldKing.Input.PlayerMovementInput>();
    input.GetType().GetField("hasFocus",flags).SetValue(input,true);
    input.GetType().GetProperty("MoveDirection").SetValue(input,Vector2.zero,null);
    System.Action moveStep=()=>{motor.GetType().GetMethod("FixedUpdate",flags).Invoke(motor,null);Physics2D.Simulate(Time.fixedDeltaTime);};
    System.Action<bool> lunge=blocked=>{
        body.position=Vector2.zero;root.transform.position=Vector3.zero;body.linearVelocity=Vector2.zero;
        thrust(Vector2.right);wall.SetActive(blocked);wall.transform.position=new Vector3(.55f,0,0);Physics2D.SyncTransforms();
        for(int i=0;i<30;i++){tick(Time.fixedDeltaTime);moveStep();}
    };
    lunge(false);check(Mathf.Abs(body.position.x-.4f)<.02f,"Motor delivers exactly the 0.4-unit lunge");
    lunge(true);check(body.position.x<.3f && body.position.x>=0,"Lunge stops at solid wall");wall.SetActive(false);
    body.position=Vector2.zero;root.transform.position=Vector3.zero;body.linearVelocity=Vector2.zero;
    thrust(Vector2.right);tick(weapon.LightDuration*.4f);weapon.ConsumeActionDisplacement();weapon.PauseOnImpact(.04f);
    tick(.02f);check(weapon.ConsumeActionDisplacement()==Vector2.zero,"Hit pause freezes lunge as well as weapon clock");
    weapon.CancelLightCombo();check(weapon.ComboTimeRemaining==0,"Interruption clears combo continuation");
    thrust(Vector2.right);player.GetType().GetMethod("OnApplicationFocus",flags).Invoke(player,new object[]{false});
    check(weapon.State==RoadOfTheOldKing.Weapons.AxeState.Held&&weapon.ComboTimeRemaining==0,"Focus loss cancels the live combo");
    player.GetType().GetField("controlsActive",flags).SetValue(player,true);
    reset();weapon.TryLightChop(Vector2.right);tick(.1f);
    root.GetComponent<RoadOfTheOldKing.Combat.Damageable>().ReceiveHit(new RoadOfTheOldKing.Combat.CombatHit(target,RoadOfTheOldKing.Combat.AttackKind.EnemyMelee,1000,Vector2.right,0f,0f));
    render();check(!surface.enabled && weapon.LightPhase==RoadOfTheOldKing.Weapons.MeleePhase.None,"Defeat cancels damage and its feedback");
} finally {
    Time.timeScale=1f;Physics2D.simulationMode=previousSimulation;
    foreach(var go in created) if(go!=null) UnityEngine.Object.DestroyImmediate(go);
    UnityEngine.Object.DestroyImmediate(upgrade);
    if(isolated!=null)UnityEngine.Object.DestroyImmediate(isolated);
}
return string.Join("\n",results.ToArray());
