// Run through Unity MCP execute_code in a fresh PrototypeLoop session with an empty,
// isolated CheckpointSession saveKey, so automatic save loading cannot skip the route.
// Resource exhaustion is covered in StaminaPlayModeChecks; these route checks refill between actions.
// Stop Play Mode afterward; these checks intentionally consume the route and defeat the player.
var results=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,name)=>results.Add((ok?"PASS ":"FAIL ")+name);
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var player=GameObject.Find("Player").GetComponent<TheLostShrine.Player.PlayerCombatController>();
var stamina=player.GetComponent<TheLostShrine.Player.PlayerStamina>();
var life=player.GetComponent<TheLostShrine.Player.PlayerHealth>();
var body=player.GetComponent<Rigidbody2D>();
var weapon=GameObject.Find("Hatchet Pickup").GetComponent<TheLostShrine.Weapons.HatchetWeapon>();
var guide=UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Prototype.PrototypeLoopGuide>();
var enemy=UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Combat.SimpleMeleeEnemy>();
var enemyHealth=enemy.GetComponent<TheLostShrine.Combat.Damageable>();
var enemyBody=enemy.GetComponent<Rigidbody2D>();
var ground=GameObject.Find("Ground").GetComponent<UnityEngine.Tilemaps.Tilemap>();
var weaponSim=weapon.GetType().GetMethod("Simulate",flags);
var guideUpdate=guide.GetType().GetMethod("Update",flags);
var enemyTick=enemy.GetType().GetMethod("Tick",flags);
System.Action<float> tickWeapon=duration=>{for(int i=0;i<Mathf.CeilToInt(duration/.02f);i++)weaponSim.Invoke(weapon,new object[]{.02f});stamina.Restore();};
System.Action tickGuide=()=>guideUpdate.Invoke(guide,null);
System.Action<float> tickEnemy=duration=>enemyTick.Invoke(enemy,new object[]{duration});
System.Action<Vector2> move=position=>{body.position=position;body.linearVelocity=Vector2.zero;player.transform.position=position;Physics2D.SyncTransforms();};
System.Action expireProtection=()=>life.GetType().GetField("invulnerableUntil",flags).SetValue(life,0f);
System.Action<UnityEngine.GameObject> expireStagger=go=>typeof(TheLostShrine.Combat.HitReaction).GetField("staggerUntil",flags).SetValue(go.GetComponent<TheLostShrine.Combat.HitReaction>(),0f);
var sceneGates=GameObject.Find("Prototype Route").GetComponentsInChildren<BoxCollider2D>(true);
System.Func<string,bool> closed=name=>System.Array.Find(sceneGates,g=>g.name==name).gameObject.activeSelf;
System.Func<Vector2,Vector2,bool> reachable=(from,to)=>{
    var start=Vector2Int.FloorToInt(from);var end=Vector2Int.FloorToInt(to);
    var seen=new System.Collections.Generic.HashSet<Vector2Int>();var queue=new System.Collections.Generic.Queue<Vector2Int>();
    queue.Enqueue(start);seen.Add(start);var hits=new System.Collections.Generic.List<Collider2D>();var filter=new ContactFilter2D{useTriggers=false};
    while(queue.Count>0){var cell=queue.Dequeue();if(cell==end)return true;
        foreach(var dir in new[]{Vector2Int.up,Vector2Int.down,Vector2Int.left,Vector2Int.right}){
            var next=cell+dir;if(seen.Contains(next)||!ground.HasTile(new Vector3Int(next.x,next.y,0)))continue;
            Physics2D.OverlapBox((Vector2)next+Vector2.one*.5f,new Vector2(.76f,.76f),0,filter,hits);
            if(hits.Exists(c=>!c.transform.IsChildOf(player.transform)))continue;
            seen.Add(next);queue.Enqueue(next);
        }
    }return false;
};
var oldSimulation=Physics2D.simulationMode;
GameObject wall=null;
try{
Physics2D.simulationMode=SimulationMode2D.Script;
check(life.Health.Health==100&&life.Health.MaxHealth==100&&life.IsAlive,"Player starts with 100 health");
check(Vector2.Dot(player.GetComponent<TheLostShrine.Player.PlayerMovement>().FacingDirection,((Vector2)weapon.transform.position-body.position).normalized)>.99f,"Player faces the hatchet placed ahead at spawn");
check(player.Weapon==null&&!player.CanRecall&&guide.Step==TheLostShrine.Prototype.PrototypeStep.Pickup,"Starts unarmed, Recall locked, pickup instruction active");
check(closed("Practice Exit Gate")&&closed("Recall Return Gate")&&closed("Enemy Approach Gate"),"All progression gates start closed");
check(!reachable(body.position,new Vector2(16.5f,-3.5f))&&!reachable(body.position,new Vector2(-17.5f,9.5f)),"Geometry prevents skipping directly to Recall or enemy");
check(enemy.State==TheLostShrine.Combat.MeleeEnemyState.Idle,"Distant enemy stays idle");
move(weapon.transform.position);Physics2D.Simulate(.02f);tickGuide();
check(player.Weapon==weapon&&guide.Step==TheLostShrine.Prototype.PrototypeStep.Chop,"Actual pickup advances to melee lesson");
check(!player.CanRecall,"Picking up the hatchet still does not grant Recall");
var dummy=GameObject.Find("Practice Dummy 1").GetComponent<TheLostShrine.Combat.Damageable>();
move((Vector2)dummy.transform.position+Vector2.down*1.2f);weapon.TryLightChop(Vector2.up);tickWeapon(.3f);tickGuide();
check(dummy.Health==90&&guide.Step==TheLostShrine.Prototype.PrototypeStep.Throw,"Real slash damage advances to throw lesson");
move((Vector2)dummy.transform.position+Vector2.down*3f);weapon.TryThrow(Vector2.up);tickWeapon(.4f);tickGuide();
check(dummy.Health==75&&weapon.State==TheLostShrine.Weapons.HatchetState.Stuck&&guide.Step==TheLostShrine.Prototype.PrototypeStep.Retrieve,"Actual throw hits dummy and teaches manual retrieval");
check(!weapon.TryRecall()&&closed("Practice Exit Gate"),"Recall and route stay locked until manual retrieval");
move(weapon.transform.position);tickWeapon(.02f);tickGuide();
check(guide.Step==TheLostShrine.Prototype.PrototypeStep.Bushes&&!closed("Practice Exit Gate"),"Retrieving opens the east practice exit");
check(!reachable(new Vector2(-3.5f,-4.5f),new Vector2(16.5f,-3.5f)),"Unbroken props physically block the route");
foreach(var position in new[]{new Vector2(-3f,-4.5f),new Vector2(0,-4.5f)}){move(position);weapon.TryLightChop(Vector2.right);tickWeapon(.3f);tickGuide();}
check(guide.Step==TheLostShrine.Prototype.PrototypeStep.Stone,"Both bushes must be cleared before the cleave lesson");
move(new Vector2(4,-4.5f));weapon.TryLightChop(Vector2.right);tickWeapon(.3f);tickGuide();
check(guide.Step==TheLostShrine.Prototype.PrototypeStep.Stone,"Ordinary slash cannot clear cracked stone");
weapon.TryBeginCharge();tickWeapon(.9f);weapon.TryReleaseCharge(Vector2.right);tickWeapon(.5f);tickGuide();
check(guide.Step==TheLostShrine.Prototype.PrototypeStep.UnlockRecall,"Full charged cleave opens the path to Recall");
check(reachable(new Vector2(-3.5f,-4.5f),new Vector2(16.5f,-3.5f)),"Cleared bush and stone pinch points fit the player collider");
check(closed("Recall Return Gate")&&closed("Enemy Approach Gate"),"Return shortcut and enemy remain locked before Recall");
move(GameObject.Find("Recall Altar").transform.position);Physics2D.Simulate(.02f);tickGuide();
check(player.CanRecall&&guide.Step==TheLostShrine.Prototype.PrototypeStep.RecallPractice&&!closed("Recall Return Gate"),"Real altar trigger grants Recall and opens the upper return route");
check(reachable(body.position,new Vector2(-12.5f,-.5f)),"Return path connects altar back to original dummy room");
check(!reachable(body.position,new Vector2(-17.5f,9.5f)),"Enemy corner stays blocked until Recall practice");
dummy.GetComponent<TheLostShrine.Prototype.PracticeTarget>().ResetTarget();
Vector2 dummyPosition=dummy.transform.position;
move(dummyPosition+new Vector2(1f,-2f));weapon.TryThrow(Vector2.up);tickWeapon(.7f);
check(weapon.State==TheLostShrine.Weapons.HatchetState.Stuck&&dummy.Health==100,"Throw beside dummy sets up a return shot");
move(dummyPosition+Vector2.down*2f);weapon.TryRecall();tickWeapon(.8f);tickGuide();
check(dummy.Health==90&&guide.Step==TheLostShrine.Prototype.PrototypeStep.Enemy&&!closed("Enemy Approach Gate"),"Real repositioned Recall damages original dummy and unlocks enemy route");
check(reachable(body.position,new Vector2(-17.5f,9.5f)),"Final enemy corner is now reachable");

// A real enemy checks the player's current position at the strike, not at wind-up.
var home=enemyBody.position;
move(home+Vector2.down*3.5f);tickEnemy(.02f);
check(enemy.State==TheLostShrine.Combat.MeleeEnemyState.Pursuing&&enemyBody.linearVelocity.y<0,"Enemy notices nearby player and approaches");
move(home+Vector2.down*1.2f);tickEnemy(.02f);
check(enemy.State==TheLostShrine.Combat.MeleeEnemyState.Windup&&enemyBody.linearVelocity==Vector2.zero,"Enemy stops and telegraphs before attacking");
tickEnemy(.3f);check(life.Health.Health==100,"Wind-up itself deals no damage");
Vector2 lockedFacing=enemy.FacingDirection;move(home+Vector2.right*1.2f);tickEnemy(.36f);tickEnemy(.02f);
check(enemy.FacingDirection==lockedFacing&&life.Health.Health==100,"Sidestepping the locked attack direction avoids damage");
tickEnemy(.15f);check(enemy.State==TheLostShrine.Combat.MeleeEnemyState.Recovering&&enemyBody.linearVelocity==Vector2.zero,"A missed strike leaves a stationary recovery opening");
tickEnemy(.4f);check(enemy.State==TheLostShrine.Combat.MeleeEnemyState.Recovering,"Recovery is long enough to counterattack");
move(home+Vector2.down*1.2f);tickEnemy(.6f);tickEnemy(.66f);tickEnemy(.02f);
check(life.Health.Health==80&&life.IsInvulnerable,"Staying in the next strike loses 20 health and grants brief immunity");
check(body.linearVelocity.y<0,"Accepted enemy strike knocks player away");
Vector2 impulse=body.linearVelocity;player.GetComponent<TheLostShrine.Player.PlayerMovement>().GetType().GetMethod("FixedUpdate",flags).Invoke(player.GetComponent<TheLostShrine.Player.PlayerMovement>(),null);
check(body.linearVelocity==impulse,"Player movement preserves knockback during stagger");
tickEnemy(.02f);check(life.Health.Health==80,"One swing cannot repeatedly damage the player");
var enemyHit=new TheLostShrine.Combat.CombatHit(enemy.gameObject,TheLostShrine.Combat.AttackKind.EnemyMelee,20,Vector2.down,0,0);
check(!life.Health.ReceiveHit(enemyHit)&&life.Health.Health==80,"Immediate follow-up hit is rejected during invulnerability");
expireProtection();check(life.Health.ReceiveHit(enemyHit)&&life.Health.Health==60,"Damage works again after protection expires");
expireProtection();expireStagger(player.gameObject);
tickEnemy(.2f);tickEnemy(1f); // Finish recovery and begin another wind-up.
wall=new GameObject("Verification occluder");wall.transform.position=home+Vector2.down*.6f;wall.AddComponent<BoxCollider2D>().size=new Vector2(2,.1f);Physics2D.SyncTransforms();
tickEnemy(.7f);tickEnemy(.02f);check(life.Health.Health==60,"Strike cannot damage through a solid wall");
tickEnemy(.2f); // Leave the occluder up for the whole active strike window.
UnityEngine.Object.DestroyImmediate(wall);wall=null;
tickEnemy(1f);
check(enemy.State==TheLostShrine.Combat.MeleeEnemyState.Windup,"Enemy can prepare another attack after recovery");
enemyHealth.ReceiveHit(new TheLostShrine.Combat.CombatHit(player.gameObject,TheLostShrine.Combat.AttackKind.LightChop,1,Vector2.up,1,.4f));tickEnemy(.8f);
check(enemy.State==TheLostShrine.Combat.MeleeEnemyState.Recovering&&life.Health.Health==60&&enemyBody.linearVelocity.y>0,"Player hit interrupts wind-up and enemy AI respects stagger/knockback");
expireStagger(enemy.gameObject);move(home+Vector2.down*9);tickEnemy(1f);tickEnemy(.02f);
check(enemy.State==TheLostShrine.Combat.MeleeEnemyState.Idle||enemy.State==TheLostShrine.Combat.MeleeEnemyState.Returning,"Enemy abandons a player beyond its home leash");
enemyHealth.ReceiveHit(new TheLostShrine.Combat.CombatHit(player.gameObject,TheLostShrine.Combat.AttackKind.ChargedCleave,1000,Vector2.up,0,0,true));tickGuide();tickEnemy(30f);
check(!enemyHealth.IsAlive&&!enemy.GetComponent<Collider2D>().enabled&&enemy.State==TheLostShrine.Combat.MeleeEnemyState.Defeated,"Defeated enemy stops attacking and stays defeated");
check(guide.Step==TheLostShrine.Prototype.PrototypeStep.Bonfire,"Defeating enemy opens the bonfire section");
int deaths=0;life.Health.Defeated+=()=>deaths++;
move(new Vector2(-14.5f,-6.5f));weapon.CancelAction();weapon.TryThrow(Vector2.right);
expireProtection();life.Health.ReceiveHit(new TheLostShrine.Combat.CombatHit(enemy.gameObject,TheLostShrine.Combat.AttackKind.EnemyMelee,1000,Vector2.down,0,0));
check(life.Health.Health==0&&!life.IsAlive&&deaths==1,"Lethal damage clamps health to zero and emits one defeat");
check(!player.enabled&&!player.GetComponent<TheLostShrine.Player.PlayerMovement>().enabled&&!body.simulated,"Death stops movement, combat input and physics");
check(weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Death cancels an in-flight hatchet without further damage");
expireProtection();check(!life.Health.ReceiveHit(enemyHit)&&deaths==1,"Dead player rejects subsequent hits");
}finally{
if(wall!=null)UnityEngine.Object.DestroyImmediate(wall);
Physics2D.simulationMode=oldSimulation;
}
return string.Join("\n",results.ToArray());
