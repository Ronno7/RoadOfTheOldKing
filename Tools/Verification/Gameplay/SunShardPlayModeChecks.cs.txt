// Run in a fresh PrototypeLoop Play Mode session with an isolated save key.
var results=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,name)=>results.Add((ok?"PASS ":"FAIL ")+name);
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var session=TheLostShrine.Progression.CheckpointSession.Instance;
string key=(string)session.GetType().GetField("saveKey",flags).GetValue(session);
if(!key.StartsWith("TheLostShrine.Verification."))throw new System.Exception("Isolated save required.");
var player=GameObject.Find("Player").GetComponent<TheLostShrine.Player.PlayerCombatController>();
var health=player.GetComponent<TheLostShrine.Player.PlayerHealth>();var body=player.GetComponent<Rigidbody2D>();
var stamina=player.GetComponent<TheLostShrine.Player.PlayerStamina>();
var interaction=player.GetComponent<TheLostShrine.Player.PlayerBonfireInteraction>();
var weapon=UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Weapons.HatchetWeapon>();
var enemy=UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Combat.SimpleMeleeEnemy>();
var enemyHealth=enemy.GetComponent<TheLostShrine.Combat.Damageable>();
var puzzle=UnityEngine.Object.FindFirstObjectByType<TheLostShrine.World.ThrowRecallPuzzle>();
var rewards=UnityEngine.Object.FindObjectsByType<TheLostShrine.World.SunShardReward>(FindObjectsSortMode.None);
var combatReward=System.Array.Find(rewards,r=>r.RewardId=="prototype/sentinel");
var puzzleReward=System.Array.Find(rewards,r=>r.RewardId=="prototype/recall-puzzle");
var exploration=System.Array.Find(rewards,r=>r.RewardId=="prototype/detour");
var camp=System.Array.Find(session.Fires,f=>!f.AllowsUpgrades);var exit=System.Array.Find(session.Fires,f=>f.AllowsUpgrades);
var tier=session.Upgrades.NextTier;var store=new TheLostShrine.Progression.PlayerPrefsProgressStore(key);
System.Action<Vector2> move=p=>{body.position=p;body.linearVelocity=Vector2.zero;player.transform.position=p;Physics2D.SyncTransforms();};
var simulation=weapon.GetType().GetMethod("Simulate",flags);
System.Action<float> tick=seconds=>{for(int i=0;i<Mathf.CeilToInt(seconds/.01f);i++)simulation.Invoke(weapon,new object[]{.01f});};
var oldSimulation=Physics2D.simulationMode;GameObject dummy=null;
try{
Physics2D.simulationMode=SimulationMode2D.Script;
check(rewards.Length==3&&combatReward!=null&&puzzleReward!=null&&exploration!=null,"Exactly three uniquely identified scene rewards");
check(session.Progress.sunShards==0&&tier.shardCost==3&&tier.choices.Length==3,"Fresh run starts empty with a three-shard, three-choice first tier");
check(!combatReward.IsAvailable&&!puzzleReward.IsAvailable&&exploration.IsAvailable,"Only exploration is available before completing encounters");
check(!exploration.TryCollect(health)&&!combatReward.TryCollect(health),"Distant and locked pickups reject collection");
player.TryEquip(weapon);player.UnlockRecall();
check(!session.TryPurchaseUpgrade(exit,tier.choices[0]),"Remote purchase is rejected");
var killingHit=new TheLostShrine.Combat.CombatHit(player.gameObject,TheLostShrine.Combat.AttackKind.ChargedCleave,1000,Vector2.up,0,0,true);
enemyHealth.ReceiveHit(killingHit);
check(combatReward.IsAvailable&&!combatReward.IsCollected&&session.Progress.sunShards==0,"Sentinel defeat drops a pickup rather than crediting currency");
check(store.Load().Has("shard/available/prototype/sentinel")&&!session.HasCheckpoint,"Uncollected drop is saved before first checkpoint");
enemy.ResetOnRest();combatReward.RestoreProgress(store.Load());
check(combatReward.IsAvailable&&!combatReward.IsCollected,"Unclaimed drop remains available after enemy reset and restoration");
move(combatReward.transform.position);check(combatReward.TryCollect(health),"Walking to the sentinel drop collects it");
check(session.Progress.sunShards==1&&store.Load().sunShards==1&&!combatReward.transform.GetChild(0).gameObject.activeSelf,"Pickup credits and saves one shard, then hides its visual");
check(!combatReward.TryCollect(health)&&session.Progress.sunShards==1,"Repeated trigger cannot duplicate a collected shard");
enemyHealth.ReceiveHit(killingHit);check(session.Progress.sunShards==1&&combatReward.IsCollected,"Killing a revived sentinel cannot farm currency");
move(camp.SpawnPosition);interaction.Open(camp);
check(!session.TryPurchaseUpgrade(camp,tier.choices[0]),"First bonfire does not sell upgrades");
interaction.Close();move(exit.SpawnPosition);interaction.Open(exit);
check(!session.TryPurchaseUpgrade(exit,tier.choices[0])&&session.Progress.sunShards==1,"Insufficient shards cannot spend or select an upgrade");
interaction.Close();
move(new Vector2(6.5f,12.5f));stamina.Restore();weapon.TryThrow(Vector2.right);tick(.7f);
move(new Vector2(6.5f,8.5f));weapon.TryRecall();tick(.8f);
check(puzzle.IsSolved&&puzzleReward.IsCollected&&session.Progress.sunShards==2,"Actual throw/reposition/Recall solves puzzle and awards one shard immediately");
puzzle.ResetOnRest();puzzle.RestoreProgress(store.Load());puzzleReward.RestoreProgress(store.Load());
check(session.Progress.sunShards==2&&puzzle.Door.IsOpen,"Restoring solved puzzle cannot duplicate its reward");
// Check the continuous walkable side route with a player-sized shape cast.
var filter=new ContactFilter2D{useTriggers=false};var hits=new System.Collections.Generic.List<RaycastHit2D>();
Vector2[] path={new Vector2(19.5f,13.5f),new Vector2(20.5f,14.5f),new Vector2(20.5f,19.5f),new Vector2(18.5f,19.5f),new Vector2(18.5f,20.5f)};
bool clear=true;move(new Vector2(100,100));
for(int i=0;i<path.Length-1;i++){var delta=path[i+1]-path[i];Physics2D.CircleCast(path[i],.3f,delta.normalized,filter,hits,delta.magnitude);if(hits.Count>0)clear=false;}
check(clear,"Player-sized path is clear from beyond the door into the exploration nook");
var ground=GameObject.Find("Ground").GetComponent<UnityEngine.Tilemaps.Tilemap>();var water=GameObject.Find("Obstacles - Water").GetComponent<UnityEngine.Tilemaps.Tilemap>();
check(ground.HasTile(new Vector3Int(18,20,0))&&!water.HasTile(new Vector3Int(18,20,0)),"Exploration shard rests on imported ground tiles, not water");
move(exploration.transform.position);check(exploration.TryCollect(health)&&session.Progress.sunShards==3,"Optional detour supplies the third shard");
move(camp.SpawnPosition);session.Rest(camp);check(session.Progress.sunShards==3&&combatReward.IsCollected&&exploration.IsCollected,"Rest keeps all shards collected");
move(exit.SpawnPosition);interaction.Open(exit);
var foreign=ScriptableObject.CreateInstance<TheLostShrine.Weapons.HatchetUpgrade>();foreign.id="foreign";
check(!session.TryPurchaseUpgrade(exit,foreign)&&session.Progress.sunShards==3,"Upgrade outside the offered tier cannot be purchased");UnityEngine.Object.DestroyImmediate(foreign);
check(session.TryPurchaseUpgrade(exit,tier.choices[0]),"Three shards purchase Quick Hands at the open second bonfire");
check(session.Progress.sunShards==0&&session.Progress.upgrades.Count==1&&session.Upgrades.NextTier==null,"Purchase spends exactly three and permanently completes this tier");
check(!session.TryPurchaseUpgrade(exit,tier.choices[1])&&!session.TryPurchaseUpgrade(exit,tier.choices[2]),"Discarded options cannot be bought after the choice");
check(Mathf.Abs(weapon.LightDuration-.2f/1.2f)<.0001f&&weapon.LightArc==110f&&Mathf.Abs(weapon.CleaveRadius-2.1f)<.001f,"Purchased Quick Hands changes only light timing");
var saved=store.Load();check(saved.sunShards==0&&saved.upgrades.Count==1&&saved.upgrades[0].upgradeId==tier.choices[0].id,"Save round trip retains spending and selected upgrade");
weapon.ApplyUpgrades(session.Upgrades.Selected);weapon.ApplyUpgrades(session.Upgrades.Selected);
check(Mathf.Abs(weapon.LightDuration-.2f/1.2f)<.0001f,"Reapplying saved upgrades does not stack the same bonus twice");
interaction.Close();move(new Vector2(100,100));weapon.CancelAction();stamina.Restore();
weapon.TryLightChop(Vector2.right);tick(.18f);check(weapon.State==TheLostShrine.Weapons.HatchetState.Held&&Mathf.Abs(stamina.Current-82)<.001f,"Faster light slash finishes earlier and still costs 18 stamina");
// Verify both geometry upgrades against real physics targets, and base settings remain unchanged.
dummy=new GameObject("Upgrade geometry target");dummy.AddComponent<CircleCollider2D>().radius=.05f;var target=dummy.AddComponent<TheLostShrine.Combat.Damageable>();
System.Action<TheLostShrine.Weapons.HatchetUpgrade> equip=u=>{weapon.CancelAction();stamina.Restore();weapon.ApplyUpgrades(u==null?new TheLostShrine.Weapons.HatchetUpgrade[0]:new[]{u});target.RestoreHealth();};
dummy.transform.position=player.transform.position+new Vector3(Mathf.Cos(68*Mathf.Deg2Rad),Mathf.Sin(68*Mathf.Deg2Rad),0);Physics2D.SyncTransforms();
equip(null);weapon.TryLightChop(Vector2.right);tick(.3f);check(target.Health==100,"Base slash misses a target outside its 110-degree arc");
equip(tier.choices[1]);weapon.TryLightChop(Vector2.right);tick(.3f);check(target.Health==90&&weapon.LightArc==150f,"Sweeping Edge hits the same target inside its 150-degree arc for normal damage");
check(Mathf.Abs(weapon.LightReach-1.35f)<.001f&&Mathf.Abs(weapon.LightDuration-.2f)<.001f,"Wider slash retains normal reach and timing");
dummy.transform.position=player.transform.position+Vector3.right*2.4f;Physics2D.SyncTransforms();
equip(null);weapon.TryBeginCharge();tick(.81f);weapon.TryReleaseCharge(Vector2.right);tick(.5f);check(target.Health==100,"Base cleave misses a target 2.4 units away");
equip(tier.choices[2]);weapon.TryBeginCharge();tick(.81f);weapon.TryReleaseCharge(Vector2.right);tick(.5f);
check(target.Health==70&&Mathf.Abs(weapon.CleaveRadius-2.6f)<.001f&&Mathf.Abs(stamina.Current-65)<.001f,"Wide Cleave reaches 2.6 units with unchanged 30 damage and 35 stamina cost");
check(weapon.Settings.lightDuration==.2f&&weapon.Settings.lightArc==110f&&weapon.Settings.cleaveRadius==2.1f,"Shared base settings remain unchanged by all upgrades");
weapon.ApplyUpgrades(session.Upgrades.Selected);
// Data-driven next tiers keep earlier selections and reject replaying old choices.
var next=ScriptableObject.CreateInstance<TheLostShrine.Weapons.HatchetUpgradeTier>();next.id="verification/tier-2";next.shardCost=4;next.choices=new[]{tier.choices[1]};
var copy=JsonUtility.FromJson<TheLostShrine.Progression.ProgressState>(JsonUtility.ToJson(session.Progress));copy.sunShards=4;
var progression=new TheLostShrine.Progression.WeaponUpgradeProgression(copy,new[]{tier,next});
check(progression.NextTier==next&&!progression.TryPurchase(tier.choices[0]),"Later tiers are reached sequentially and reject prior-tier choices");
check(progression.TryPurchase(tier.choices[1])&&copy.upgrades.Count==2&&copy.sunShards==0,"A future data asset tier can add another permanent choice at its own cost");
weapon.ApplyUpgrades(progression.Selected);check(weapon.LightArc==150&&Mathf.Abs(weapon.LightDuration-.2f/1.2f)<.001f,"Different tier effects compound on the same weapon");UnityEngine.Object.DestroyImmediate(next);weapon.ApplyUpgrades(session.Upgrades.Selected);
string legacy=key+".legacy";PlayerPrefs.SetString(legacy,"{\"version\":1,\"hasHatchet\":true,\"recallUnlocked\":true,\"checkpointId\":\"\",\"scenePath\":\"\",\"discoveredFires\":[],\"completedIds\":[]}");
var migrated=new TheLostShrine.Progression.PlayerPrefsProgressStore(legacy).Load();check(migrated.hasHatchet&&migrated.sunShards==0&&migrated.upgrades!=null&&migrated.upgrades.Count==0,"Pre-upgrade version-one saves load without losing existing progress");PlayerPrefs.DeleteKey(legacy);
move(exit.SpawnPosition);interaction.Open(exit);
check(session.Progress.sunShards==0&&session.Progress.upgrades.Count==1,"Rest retains the purchased upgrade and spent balance");
}finally{if(dummy!=null)UnityEngine.Object.DestroyImmediate(dummy);Physics2D.simulationMode=oldSimulation;}
return string.Join("\n",results.ToArray());
