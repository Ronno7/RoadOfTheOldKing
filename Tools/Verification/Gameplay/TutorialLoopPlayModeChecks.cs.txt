// Run once per phase in Tutorial Play Mode, using a save key prefixed
// TheLostShrine.Verification. Phase 0/1/3 reload the scene; wait between calls.
// After phase 2, leave and re-enter Play Mode with the same isolated key for phase 3.
if (!EditorApplication.isPlaying) throw new System.Exception("Play Mode required.");
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic;
var session = TheLostShrine.Progression.CheckpointSession.Instance;
if (session == null) throw new System.Exception("Tutorial session missing.");
string key = (string)session.GetType().GetField("saveKey", flags).GetValue(session);
if (!key.StartsWith("TheLostShrine.Verification.")) throw new System.Exception("Isolated save required.");
var results = new System.Collections.Generic.List<string>();
System.Action<bool, string> check = (ok, name) => {
    if (!ok) throw new System.Exception("FAIL: " + name);
    results.Add("PASS: " + name);
};
var player = UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Player.PlayerHealth>();
var combat = player.GetComponent<TheLostShrine.Player.PlayerCombatController>();
var body = player.GetComponent<Rigidbody2D>();
var stamina = player.GetComponent<TheLostShrine.Player.PlayerStamina>();
var interaction = player.GetComponent<TheLostShrine.Player.PlayerBonfireInteraction>();
var enemy = UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Combat.SimpleMeleeEnemy>();
var enemyHealth = enemy.GetComponent<TheLostShrine.Combat.Damageable>();
var reward = UnityEngine.Object.FindFirstObjectByType<TheLostShrine.World.SunShardReward>();
var fire = session.Fires.Single();
var store = new TheLostShrine.Progression.PlayerPrefsProgressStore(key);
var scene = UnityEngine.SceneManagement.SceneManager.GetActiveScene();
check(scene.path == "Assets/Scenes/Tutorial.unity" && scene.buildIndex >= 0, "Tutorial supports scene reload");
check(reward.RewardId == "tutorial/first-enemy" && fire.Id == "tutorial/roadside-fire", "Stable Tutorial reward/checkpoint IDs");
int phase = SessionState.GetInt("Tutorial.LoopCheckPhase", 0);
System.Action<Vector2> move = p => {
    body.position = p; body.linearVelocity = Vector2.zero;
    player.transform.position = p; Physics2D.SyncTransforms();
};
var killingHit = new TheLostShrine.Combat.CombatHit(player.gameObject,
    TheLostShrine.Combat.AttackKind.ChargedCleave, 1000, Vector2.up, 0, 0, true);
System.Action die = () => {
    player.Health.ReceiveHit(new TheLostShrine.Combat.CombatHit(enemy.gameObject,
        TheLostShrine.Combat.AttackKind.EnemyMelee, 1000, Vector2.down, 0, 0));
    check(!player.IsAlive, "Player defeat");
    session.Respawn();
};
if (phase == 0) {
    check(!session.HasCheckpoint && session.Progress.sunShards == 0 && !reward.IsAvailable, "Fresh run has no checkpoint or reward");
    check(Vector2.Distance(player.transform.position, new Vector2(15, 13)) < .1f && !combat.CanRecall, "Home start; Recall locked");
    check(!session.Rest(fire) && !reward.TryCollect(player), "Remote rest and locked pickup rejected");
    var weapon = UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Weapons.HatchetWeapon>();
    check(combat.TryEquip(weapon), "Existing hatchet equips");
    enemyHealth.ReceiveHit(killingHit);
    check(reward.IsAvailable && !reward.IsCollected && session.Progress.sunShards == 0, "Defeat unlocks a physical shard");
    check(store.Load().Has("shard/available/tutorial/first-enemy"), "Uncollected drop saved before first rest");
    SessionState.SetInt("Tutorial.LoopCheckPhase", 1);
    die();
} else if (phase == 1) {
    check(player.IsAlive && !session.HasCheckpoint && Vector2.Distance(player.transform.position, new Vector2(15,13)) < .1f, "Death before rest returns home");
    check(combat.Weapon != null && !combat.CanRecall, "Hatchet ownership restored without unlocking Recall");
    check(reward.IsAvailable && !reward.IsCollected && enemyHealth.IsAlive, "Uncollected shard survives reload and enemy respawn");
    move(reward.transform.position);
    check(reward.TryCollect(player) && session.Progress.sunShards == 1, "Pickup credits exactly one shard");
    check(!reward.TryCollect(player) && store.Load().sunShards == 1, "Duplicate pickup rejected and balance saved");
    move(fire.SpawnPosition);
    player.Health.ReceiveHit(new TheLostShrine.Combat.CombatHit(enemy.gameObject,
        TheLostShrine.Combat.AttackKind.EnemyMelee, 10, Vector2.up, 0, 0));
    stamina.TrySpend(50);
    check(interaction.Open(fire), "Nearby fire opens and rests");
    check(player.Health.Health == player.Health.MaxHealth && stamina.Current == stamina.Maximum, "Rest restores health and stamina");
    check(session.HasCheckpoint && fire.IsDiscovered && enemyHealth.IsAlive, "Rest saves checkpoint and resets enemy");
    check(!player.GetComponent<TheLostShrine.Input.PlayerMovementInput>().enabled &&
          !player.GetComponent<TheLostShrine.Input.PlayerCombatInput>().enabled, "Fire menu suspends action input");
    interaction.Close();
    check(player.GetComponent<TheLostShrine.Input.PlayerMovementInput>().enabled &&
          player.GetComponent<TheLostShrine.Input.PlayerCombatInput>().enabled, "Closing fire restores input");
    SessionState.SetInt("Tutorial.LoopCheckPhase", 2);
    die();
} else if (phase == 2 || phase == 3) {
    check(player.IsAlive && Vector2.Distance(player.transform.position, fire.SpawnPosition) < .1f, phase == 2 ? "Death after rest returns to bonfire" : "New Play session restores checkpoint");
    check(session.Progress.sunShards == 1 && reward.IsCollected, "Collected reward stays collected");
    enemyHealth.ReceiveHit(killingHit);
    move(reward.transform.position);
    check(!reward.TryCollect(player) && session.Progress.sunShards == 1, "Respawned enemy cannot farm shards");
    move(fire.SpawnPosition);
    check(session.Rest(fire) && enemyHealth.IsAlive && session.Progress.sunShards == 1, "Repeated rest retains progression");
    SessionState.SetInt("Tutorial.LoopCheckPhase", phase + 1);
    if (phase == 3) session.StartNewRun();
} else if (phase == 4) {
    check(!session.HasCheckpoint && session.Progress.sunShards == 0 && !reward.IsAvailable && !combat.CanRecall, "New run clears only test progression");
    check(Vector2.Distance(player.transform.position, new Vector2(15,13)) < .1f, "New run starts at home");
    SessionState.SetInt("Tutorial.LoopCheckPhase", 5);
} else throw new System.Exception("All phases already completed.");
return new { phase, checks = results, next = phase == 2 ? "Exit/re-enter Play Mode with same test key" : "Wait for any reload, then run next phase" };
