var results=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,name)=>results.Add((ok?"PASS ":"FAIL ")+name);
var player=GameObject.Find("Player").GetComponent<TheLostShrine.Player.PlayerCombatController>();
var weapon=GameObject.Find("Hatchet Pickup").GetComponent<TheLostShrine.Weapons.HatchetWeapon>();
var body=player.GetComponent<Rigidbody2D>();
var sim=typeof(TheLostShrine.Weapons.HatchetWeapon).GetMethod("Simulate",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic);
System.Action<float> tick=duration=>{int steps=Mathf.CeilToInt(duration/.02f);for(int i=0;i<steps;i++)sim.Invoke(weapon,new object[]{.02f});};
System.Action<Vector2> move=position=>{body.position=position;body.linearVelocity=Vector2.zero;player.transform.position=position;Physics2D.SyncTransforms();};
var created=new System.Collections.Generic.List<GameObject>();
System.Func<string,Vector2,GameObject> spawn=(path,position)=>{var go=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>(path));go.name="Verification Target";go.transform.position=position;var rb=go.GetComponent<Rigidbody2D>();if(rb!=null)rb.position=position;created.Add(go);Physics2D.SyncTransforms();return go;};
string dummyPath="Assets/Prefabs/Combat/PracticeDummy.prefab";
var oldSimulation=Physics2D.simulationMode;
var start=body.position;
try{
Physics2D.simulationMode=SimulationMode2D.Script;
check(player.Weapon==null && weapon.State==TheLostShrine.Weapons.HatchetState.OnGround,"Starts unarmed with grounded prefab");
check(!weapon.TryThrow(Vector2.right) && !weapon.TryLightChop(Vector2.right),"Ground weapon rejects attacks");
move(weapon.transform.position);Physics2D.Simulate(.02f);
check(player.Weapon==weapon && weapon.State==TheLostShrine.Weapons.HatchetState.Held && !weapon.GetComponent<Collider2D>().enabled,"Real trigger picks up the existing hatchet");
check(!player.CanRecall,"Pickup leaves Recall locked");
check(!player.TryEquip(weapon),"Duplicate equip rejected");
Vector2 origin=new Vector2(-4,-6);move(origin);
var dummy=spawn(dummyPath,origin+Vector2.right);
var health=dummy.GetComponent<TheLostShrine.Combat.Damageable>();
var extra=new GameObject("Extra hit collider");extra.transform.SetParent(dummy.transform,false);extra.AddComponent<CircleCollider2D>().radius=.25f;
for(int i=0;i<3;i++){check(weapon.TryLightChop(Vector2.right) && weapon.ComboIndex==i,"Combo stage "+(i+1));check(!weapon.TryThrow(Vector2.up) && !weapon.TryBeginCharge(),"Action exclusivity during chop "+(i+1));tick(.34f);}
check(health.Health==4,"Three-hit combo deals 1/1/2, once per target despite multiple colliders");
tick(.7f);check(weapon.TryLightChop(Vector2.right) && weapon.ComboIndex==0,"Combo resets after timeout");tick(.34f);
dummy.SetActive(false);
var behind=spawn(dummyPath,origin+Vector2.left);
weapon.TryLightChop(Vector2.right);tick(.34f);
check(behind.GetComponent<TheLostShrine.Combat.Damageable>().Health==8,"Directional chop excludes targets behind");
weapon.TryBeginCharge();tick(.1f);check(!weapon.TryReleaseCharge(Vector2.right) && weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Tap below minimum charge cancels");
weapon.TryBeginCharge();tick(.9f);check(weapon.Charge01==1f,"Charge clamps at full strength");
weapon.TryReleaseCharge(Vector2.right);tick(.48f);
check(behind.GetComponent<TheLostShrine.Combat.Damageable>().Health==4,"Full cleave hits behind for four damage");
check(behind.GetComponent<TheLostShrine.Combat.HitReaction>().IsStaggered && behind.GetComponent<Rigidbody2D>().linearVelocity.x<0,"Cleave applies outward knockback and stagger");
behind.SetActive(false);
var guarded=spawn("Assets/Prefabs/Combat/ShieldPracticeDummy.prefab",origin+Vector2.right);
var guardHealth=guarded.GetComponent<TheLostShrine.Combat.Damageable>();var shield=guarded.GetComponent<TheLostShrine.Combat.ShieldProtection>();
weapon.TryLightChop(Vector2.right);tick(.34f);
weapon.TryBeginCharge();tick(.3f);weapon.TryReleaseCharge(Vector2.right);tick(.48f);
check(guardHealth.Health==8 && shield.IsIntact,"Shield blocks light and partial cleave");
weapon.TryBeginCharge();tick(.9f);weapon.TryReleaseCharge(Vector2.right);tick(.48f);
check(guardHealth.Health==4 && !shield.IsIntact,"Full cleave breaks shield and damages target");
guarded.SetActive(false);
var bush=spawn("Assets/Prefabs/World/Bush.prefab",origin+Vector2.right);
weapon.TryLightChop(Vector2.right);tick(.34f);
check(!bush.activeSelf,"Light chop breaks bushes");
var stone=spawn("Assets/Prefabs/World/CrackedStone.prefab",origin+Vector2.right);
weapon.TryLightChop(Vector2.right);tick(.34f);check(stone.activeSelf,"Cracked stone resists light chop");
weapon.TryBeginCharge();tick(.9f);weapon.TryReleaseCharge(Vector2.right);tick(.48f);check(!stone.activeSelf,"Full cleave breaks cracked stone");
weapon.TryThrow(Vector2.right);check(!weapon.TryRecall(),"Locked Recall rejects in-flight recall");
check(!weapon.TryLightChop(Vector2.right) && !weapon.TryBeginCharge(),"Melee unavailable while weapon is away");tick(.7f);
check(weapon.State==TheLostShrine.Weapons.HatchetState.Stuck && Mathf.Abs(Vector2.Distance(origin,weapon.transform.position)-6)<.05f,"Throw stops at configured range");
check(!weapon.TryRecall(),"Locked Recall rejects stuck recall");
move(weapon.transform.position);tick(.02f);check(weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Walk-up retrieval works before Recall unlock");
move(new Vector2(-1,7));Physics2D.Simulate(.02f);check(player.CanRecall,"Separate altar trigger unlocks Recall");
move(origin);
var obstacle=new GameObject("Verification wall");obstacle.transform.position=origin+Vector2.right*2;obstacle.AddComponent<BoxCollider2D>().size=new Vector2(.2f,2);created.Add(obstacle);Physics2D.SyncTransforms();
weapon.TryThrow(Vector2.right);tick(.3f);check(weapon.State==TheLostShrine.Weapons.HatchetState.Stuck && weapon.transform.position.x<origin.x+1.8f,"Swept throw stops at solid obstacle");
weapon.TryRecall();tick(.5f);check(weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Recall returns from obstacle");
var blockedTarget=spawn(dummyPath,origin+Vector2.right*2.7f);
var detector=new TheLostShrine.Weapons.HatchetHitDetector(player.transform,weapon.transform);
detector.Melee(origin,Vector2.right,4,120,new TheLostShrine.Combat.CombatHit(player.gameObject,TheLostShrine.Combat.AttackKind.LightChop,1,Vector2.right,0,0));
check(blockedTarget.GetComponent<TheLostShrine.Combat.Damageable>().Health==8,"Melee cannot hit through solid terrain");
obstacle.SetActive(false);blockedTarget.SetActive(false);
var flightTarget=spawn(dummyPath,origin+Vector2.right*2);
weapon.TryThrow(Vector2.right);tick(.22f);check(flightTarget.GetComponent<TheLostShrine.Combat.Damageable>().Health==6 && weapon.State==TheLostShrine.Weapons.HatchetState.Stuck,"Outbound throw damages once and sticks to target");
weapon.TryRecall();check(!weapon.TryRecall(),"Repeated recall cannot restart return");tick(.4f);
check(flightTarget.GetComponent<TheLostShrine.Combat.Damageable>().Health==4,"Same target can be hit once outbound and once returning");
flightTarget.SetActive(false);
weapon.TryThrow(Vector2.right);tick(.7f);
var returnA=spawn(dummyPath,origin+Vector2.right*2);var returnB=spawn(dummyPath,origin+Vector2.right*4);
obstacle.SetActive(true);Physics2D.SyncTransforms();
weapon.TryRecall();tick(.14f);move(origin+Vector2.down*.2f);tick(.6f);
check(returnA.GetComponent<TheLostShrine.Combat.Damageable>().Health==6 && returnB.GetComponent<TheLostShrine.Combat.Damageable>().Health==6,"Returning hatchet pierces multiple targets once each");
check(weapon.State==TheLostShrine.Weapons.HatchetState.Held && Vector2.Distance(weapon.transform.position,body.position)<.1f,"Return crosses terrain and tracks moving owner");
returnA.SetActive(false);returnB.SetActive(false);obstacle.SetActive(false);
move(origin);var doomed=spawn(dummyPath,origin+Vector2.right*2);
weapon.TryThrow(Vector2.right);tick(.22f);UnityEngine.Object.DestroyImmediate(doomed);tick(.1f);
check(weapon!=null && weapon.State==TheLostShrine.Weapons.HatchetState.Stuck,"Destroying struck target preserves the hatchet");
weapon.TryRecall();tick(.5f);
check(UnityEngine.Object.FindObjectsByType<TheLostShrine.Weapons.HatchetWeapon>(FindObjectsSortMode.None).Length==1,"Single weapon instance throughout lifecycle");
}finally{
foreach(var go in created)if(go!=null)UnityEngine.Object.DestroyImmediate(go);
move(start);Physics2D.simulationMode=oldSimulation;
}
return string.Join("\n",results.ToArray());
