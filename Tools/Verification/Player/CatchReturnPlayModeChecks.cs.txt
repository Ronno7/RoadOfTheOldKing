// Empty Play Mode scene, real prefabs. Verify event ownership, not just a timer.
if(!Application.isPlaying || UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Progression.CheckpointSession>()!=null)
    throw new Exception("Use an empty Play Mode scene without a checkpoint session.");
var results=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,label)=>results.Add((ok?"PASS ":"FAIL ")+label);
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var objects=new System.Collections.Generic.List<GameObject>();
var oldSimulation=Physics2D.simulationMode;
System.Func<string,GameObject> spawn=path=>{var go=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>(path));objects.Add(go);return go;};
try {
    Physics2D.simulationMode=SimulationMode2D.Script;
    var player=spawn("Assets/Prefabs/Player/Player.prefab").GetComponent<TheLostShrine.Player.PlayerCombatController>();
    var weapon=spawn("Assets/Prefabs/Weapons/Hatchet.prefab").GetComponent<TheLostShrine.Weapons.HatchetWeapon>();
    var body=player.GetComponent<Rigidbody2D>();var stamina=player.GetComponent<TheLostShrine.Player.PlayerStamina>();
    player.transform.position=new Vector3(100,100,0);body.position=player.transform.position;
    int catches=0;bool stateCorrect=true;
    var art=AssetDatabase.LoadAssetAtPath<TheLostShrine.Player.RegisteredActionSprites>(TheLostShrine.Editor.RegisteredSpriteProofBuilder.CatchAssetPath);
    var clock=new TheLostShrine.Player.CatchPresentationClock();
    weapon.ReturnedToHand+=()=>{catches++;stateCorrect &= weapon.State==TheLostShrine.Weapons.HatchetState.Held;clock.ConfirmCatch();};
    System.Action<float> tick=dt=>weapon.GetType().GetMethod("Simulate",flags).Invoke(weapon,new object[]{dt});
    check(player.TryEquip(weapon) && catches==0,"Equip does not send a Recall catch");
    player.UnlockRecall();
    check(weapon.TryThrow(Vector2.right),"Quick throw begins on the real weapon");
    tick(.4f);clock.BeginReach(art);
    check(weapon.TryRecall(),"Real Recall begins");
    tick(.001f);clock.Advance(100);
    check(catches==0 && !clock.HasCaught && clock.CelIndex==0,"Reach waits while the actual weapon is still returning");
    tick(1);
    check(catches==1 && stateCorrect && clock.HasCaught && clock.CelIndex==1,"Arrival confirms ownership before notifying the catch clock");
    tick(1);tick(1);
    check(catches==1,"Held updates never repeat the arrival event");
    stamina.Restore();weapon.TryThrow(Vector2.right);tick(.4f);weapon.TryRecall();clock.BeginReach(art);
    weapon.CancelAction();clock.Cancel();tick(1);
    check(catches==1 && !clock.HasCaught,"Cancelled return produces no catch");
    stamina.Restore();weapon.TryThrow(Vector2.right);tick(1);
    check(weapon.State==TheLostShrine.Weapons.HatchetState.Stuck,"Unrecalled throw reaches its range limit");
    player.transform.position=weapon.transform.position;body.position=player.transform.position;Physics2D.SyncTransforms();tick(.02f);
    check(weapon.State==TheLostShrine.Weapons.HatchetState.Held && catches==1,"Walking over a grounded axe does not masquerade as an airborne catch");
    stamina.Restore();weapon.TryThrow(Vector2.right);tick(1);
    player.transform.position+=(Vector3)(Vector2.left*(weapon.Settings.autoRecallDistance+1));body.position=player.transform.position;
    Physics2D.SyncTransforms();clock.BeginReach(art);tick(.01f);
    check(weapon.State==TheLostShrine.Weapons.HatchetState.Returning,"Distance-triggered Recall uses the same return path");
    tick(4);
    check(catches==2 && clock.HasCaught,"Automatic Recall sends one confirmed arrival");
    clock.Advance(.23f);
    check(!clock.IsActive && weapon.TryLightChop(Vector2.right),"Catch presentation adds no gameplay attack lock");
}
finally {
    Physics2D.simulationMode=oldSimulation;
    foreach(var go in objects) if(go!=null)UnityEngine.Object.DestroyImmediate(go);
}
return string.Join("\n",results.ToArray());
