// Empty unsaved Play Mode scene. Run once per direction with real prefabs/motor, no save session.
int direction = 0; // 0 east, 1 north, 2 south, 3 west
Vector2 aim=new[]{Vector2.right,Vector2.up,Vector2.down,Vector2.left}[direction];
string facing=new[]{"East","North","South","West"}[direction];
var moveKey=new[]{UnityEngine.InputSystem.Key.D,UnityEngine.InputSystem.Key.W,UnityEngine.InputSystem.Key.S,UnityEngine.InputSystem.Key.A}[direction];
if(!Application.isPlaying || UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Progression.CheckpointSession>()!=null)
    throw new Exception("Use an empty Play Mode scene without a save session.");
var results=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,name)=>results.Add((ok?"PASS ":"FAIL ")+name);
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var root=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Player/Player.prefab"));
var prop=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Weapons/Hatchet.prefab"));
var combat=root.GetComponent<TheLostShrine.Player.PlayerCombatController>();
var weapon=prop.GetComponent<TheLostShrine.Weapons.HatchetWeapon>();
var movement=root.GetComponent<TheLostShrine.Player.PlayerMovement>();
var motor=root.GetComponent<Rigidbody2D>();motor.interpolation=RigidbodyInterpolation2D.None;
var dash=root.GetComponent<TheLostShrine.Player.PlayerDash>();
var stamina=root.GetComponent<TheLostShrine.Player.PlayerStamina>();
var life=root.GetComponent<TheLostShrine.Player.PlayerHealth>();
var presenter=root.GetComponent<TheLostShrine.Player.RegisteredPlayerAnimation>();
var animation=root.GetComponent<TheLostShrine.Player.PlayerLocomotionAnimation>();
var input=root.GetComponent<TheLostShrine.Input.PlayerMovementInput>();
var view=weapon.GetComponent<TheLostShrine.Weapons.HatchetView>();
var body=root.transform.Find("Body").GetComponent<SpriteRenderer>();
var held=body.transform.Find("Drawn held axe").GetComponent<SpriteRenderer>();
var blade=prop.transform.Find("Model/Sprite").GetComponent<SpriteRenderer>();
var art=AssetDatabase.LoadAssetAtPath<TheLostShrine.Player.RegisteredActionSprites>(TheLostShrine.Editor.RegisteredSpriteProofBuilder.DashAssetPath);
var oldSim=Physics2D.simulationMode;
var oldRouting=UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode;
var keyboard=UnityEngine.InputSystem.InputSystem.AddDevice<UnityEngine.InputSystem.Keyboard>();
var actions=(UnityEngine.InputSystem.InputActionMap)input.GetType().GetField("actions",flags).GetValue(input);
actions.devices=new UnityEngine.InputSystem.InputDevice[]{keyboard};
GameObject wall=null;
System.Action draw=()=>{animation.GetType().GetMethod("Present",flags).Invoke(animation,new object[]{.02f});view.GetType().GetMethod("LateUpdate",flags).Invoke(view,null);};
System.Action<UnityEngine.InputSystem.Key[]> keys=heldKeys=>{
    UnityEngine.InputSystem.InputSystem.QueueStateEvent(keyboard,new UnityEngine.InputSystem.LowLevel.KeyboardState(heldKeys));
    UnityEngine.InputSystem.InputSystem.Update();input.GetType().GetField("hasFocus",flags).SetValue(input,true);
    input.GetType().GetMethod("Update",flags).Invoke(input,null);
};
System.Action step=()=>{movement.GetType().GetMethod("FixedUpdate",flags).Invoke(movement,null);Physics2D.Simulate(.02f);draw();};
System.Action<int> steps=count=>{for(int i=0;i<count;i++)step();};
System.Action reset=()=>{
    weapon.CancelAction();life.HealAtRest();movement.enabled=true;combat.enabled=true;
    combat.GetType().GetMethod("ProcessInput",flags).Invoke(combat,new object[]{new TheLostShrine.Input.CombatInputFrame{Active=true}});
    keys(new UnityEngine.InputSystem.Key[0]);motor.position=new Vector2(200,200);root.transform.position=motor.position;
    motor.linearVelocity=Vector2.zero;Physics2D.SyncTransforms();draw();
};
System.Action start=()=>{keys(new[]{moveKey,UnityEngine.InputSystem.Key.Space});step();keys(new UnityEngine.InputSystem.Key[0]);};
try {
    Physics2D.simulationMode=SimulationMode2D.Script;
    UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode=UnityEngine.InputSystem.InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;
    input.GetType().GetField("paused",flags).SetValue(input,false);
    string error;check(art.IsRegistered(out error) && art.celCount==5 && art.exposures.Length==5 && art.views.Length==4,"All four cardinal views have five paired dash cels with common foot pivot/PPU");
    check(new SerializedObject(presenter).FindProperty("aimGaits").objectReferenceValue==null,"Rejected aim-walk frames are detached from the runtime prefab");
    reset();start();
    check(combat.Weapon==null && presenter.DisplayedAction=="Dash" && presenter.DisplayedCel==0 && !held.enabled,"Push-off works before pickup without a phantom axe");
    check(dash.HasDodgeProtection && Mathf.Abs(stamina.Current-75)<.001f,"Accepted dash retains 25 stamina cost and opening dodge protection");
    int cel=presenter.DisplayedCel;for(int i=0;i<20;i++)draw();
    check(presenter.DisplayedCel==cel && dash.Progress==0,"Render samples cannot advance the motor-owned dash clock");
    steps(2);check(presenter.DisplayedCel==1,"Drive follows the first 40 ms of gameplay dash");
    steps(4);check(presenter.DisplayedCel==2 && !dash.HasDodgeProtection,"Travel pose continues after the original 100 ms dodge window");
    steps(2);check(dash.IsDashing && Mathf.Abs(Vector2.Distance(motor.position,new Vector2(200,200))-2.2f)<.025f,"Real motor covers exactly 2.2 units in the existing dash travel");
    steps(2);check(!dash.IsDashing && presenter.DisplayedCel==3 && dash.CooldownRemaining<=.351f && dash.CooldownRemaining>=.329f,"Natural completion begins landing with the existing motor cooldown");
    steps(3);check(presenter.DisplayedAction=="Dash" && presenter.DisplayedCel==4,"Recovery settles through the fifth pose");
    steps(5);check(!presenter.IsPresentingBody && body.sprite.name==facing+"_00" && motor.linearVelocity==Vector2.zero,"Recovery returns to original idle with no extra movement");

    reset();check(combat.TryEquip(weapon),"Real Tutorial hatchet equips");start();
    check(presenter.DisplayedAction=="Dash" && held.enabled && held.sprite==art.views[direction].weapon[0] && !blade.enabled,"Equipped dash pairs its body and axe, with exactly one visible weapon");
    dash.Cancel();draw();check(!presenter.IsPresentingBody && blade.enabled && !held.enabled,"Early cancellation clears the dash instead of playing a false landing");
    reset();start();steps(10);keys(new[]{moveKey});step();
    check(!presenter.IsPresentingBody && Vector2.Distance(motor.linearVelocity,aim*movement.MoveSpeed)<.001f && blade.enabled,"Movement immediately replaces recovery at full walk speed");
    keys(new UnityEngine.InputSystem.Key[0]);step();check(!presenter.IsPresentingBody,"Stopping cannot replay interrupted recovery");
    reset();start();steps(10);check(weapon.TryLightChop(aim),"Landing introduces no attack lock");draw();
    check(presenter.DisplayedAction=="Forehand" && held.enabled && !blade.enabled,"Immediate attack replaces dash recovery using its authored cardinal view");
    reset();start();steps(10);check(weapon.TryBeginThrow(aim),"Landing introduces no throw lock");draw();
    check(presenter.DisplayedAction=="Throw" && weapon.IsThrowing && !blade.enabled && held.enabled,"New throw overrides recovery using its authored cardinal view");
    reset();weapon.TryBeginThrow(aim);weapon.GetType().GetMethod("Simulate",flags).Invoke(weapon,new object[]{.09f});draw();start();
    check(!weapon.IsThrowing && presenter.DisplayedAction=="Dash" && stamina.Current==75,"Accepted dash cancels uncommitted aim once and shows the paired dash");
    reset();stamina.TrySpend(100);start();check(!dash.IsDashing && presenter.DisplayedAction!="Dash" && stamina.Current==0,"Unaffordable dash cannot start a visual sequence");

    reset();wall=new GameObject("Dash verification wall");wall.layer=LayerMask.NameToLayer("Environment");wall.transform.position=new Vector3(200,200,0)+(Vector3)aim*1.5f;wall.AddComponent<BoxCollider2D>().size=Mathf.Abs(aim.x)>.5f?new Vector2(1,8):new Vector2(8,1);Physics2D.SyncTransforms();
    start();steps(15);check(Vector2.Dot(motor.position-new Vector2(200,200),aim)<.64f && !dash.IsDashing && !dash.HasDodgeProtection,"Wall collision stops travel and cannot prolong dash or immunity");
    UnityEngine.Object.DestroyImmediate(wall);wall=null;
    reset();start();movement.enabled=false;draw();check(!presenter.IsPresentingBody && !held.enabled,"Disabling the motor clears dash visuals immediately");
    reset();start();combat.enabled=false;draw();check(!presenter.IsPresentingBody && !held.enabled,"Control loss clears dash and its held layer");
    reset();start();motor.position+=Vector2.right*10;root.transform.position=motor.position;draw();
    check(!presenter.IsPresentingBody,"Teleport suppresses the old dash presentation");
    step();check(!presenter.IsPresentingBody,"Suppressed dash cannot restart after teleport");
    reset();keys(new[]{UnityEngine.InputSystem.Key.A,UnityEngine.InputSystem.Key.Space});step();
    check(dash.IsDashing && presenter.DisplayedAction=="Dash" && body.sprite==art.views[3].body[0] && held.sprite==art.views[3].weapon[0] && !body.flipX,"West dash selects its own authored body/axe pair without mirroring");
    bool diagonalViews=true,diagonalTravel=true;
    var horizontal=new[]{UnityEngine.InputSystem.Key.D,UnityEngine.InputSystem.Key.A};
    var vertical=new[]{UnityEngine.InputSystem.Key.W,UnityEngine.InputSystem.Key.S};
    for(int h=0;h<2;h++)for(int v=0;v<2;v++) {
        reset();keys(new[]{horizontal[h],vertical[v],UnityEngine.InputSystem.Key.Space});step();keys(new UnityEngine.InputSystem.Key[0]);
        diagonalViews&=dash.IsDashing && presenter.DisplayedAction=="Dash" && body.sprite==art.views[h==0?0:3].body[0] && !body.flipX;
        steps(8);Vector2 travelled=motor.position-new Vector2(200,200);
        diagonalTravel&=Mathf.Abs(travelled.magnitude-2.2f)<.025f && Mathf.Abs(Mathf.Abs(travelled.x)-Mathf.Abs(travelled.y))<.005f;
    }
    check(diagonalViews,"All four diagonal inputs select the matching authored side view");
    check(diagonalTravel,"Diagonal dashes retain normalized 2.2-unit travel without a speed bonus");
    reset();start();input.GetType().GetMethod("OnApplicationFocus",flags).Invoke(input,new object[]{false});step();
    check(!dash.IsDashing && !presenter.IsPresentingBody && motor.linearVelocity==Vector2.zero,"Focus loss stops dash movement and presentation");

    reset();weapon.TryThrow(aim);weapon.GetType().GetMethod("Simulate",flags).Invoke(weapon,new object[]{.4f});draw();start();
    check(weapon.IsAway && presenter.DisplayedAction=="Dash" && !held.enabled && blade.enabled,"Dash during flight shows empty hands and preserves the single world axe");
    check(body.transform.localScale==Vector3.one && body.transform.localPosition==new Vector3(0,-.25f,0) && !body.flipX,"Dash leaves body scale, registration and asymmetric facing intact");
    dash.Cancel();draw();check(blade.enabled && !held.enabled,"Flight remains visible after dash cancellation");
}
finally {
    if(wall!=null)UnityEngine.Object.DestroyImmediate(wall);
    Physics2D.simulationMode=oldSim;
    UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode=oldRouting;
    UnityEngine.InputSystem.InputSystem.RemoveDevice(keyboard);
    UnityEngine.Object.DestroyImmediate(root);UnityEngine.Object.DestroyImmediate(prop);
}
return new{passed=results.Count(s=>s.StartsWith("PASS")),failed=results.Count(s=>s.StartsWith("FAIL")),results};
