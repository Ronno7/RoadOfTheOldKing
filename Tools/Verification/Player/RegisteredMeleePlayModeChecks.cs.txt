// Empty, unsaved Play Mode scene. Real Tutorial prefabs; no checkpoint/save session.
int direction = 0; // 0 east, 1 north, 2 south, 3 west
Vector2 aim=new[]{Vector2.right,Vector2.up,Vector2.down,Vector2.left}[direction];
if (!Application.isPlaying || UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Progression.CheckpointSession>() != null)
    throw new Exception("Use an empty Play Mode scene without a checkpoint session.");
var results=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,label)=>results.Add((ok?"PASS ":"FAIL ")+label);
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var created=new System.Collections.Generic.List<GameObject>();
System.Func<string,GameObject> spawn=path=>{var go=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>(path));created.Add(go);return go;};
var oldSimulation=Physics2D.simulationMode;
var upgrade=ScriptableObject.CreateInstance<TheLostShrine.Weapons.HatchetUpgrade>();
try {
    Physics2D.simulationMode=SimulationMode2D.Script;
    var player=spawn("Assets/Prefabs/Player/TutorialPlayer.prefab").GetComponent<TheLostShrine.Player.PlayerCombatController>();
    var weapon=spawn("Assets/Prefabs/Weapons/TutorialHatchet.prefab").GetComponent<TheLostShrine.Weapons.HatchetWeapon>();
    var target=spawn("Assets/Prefabs/Combat/PracticeDummy.prefab");
    var hp=target.GetComponent<TheLostShrine.Combat.Damageable>();
    var stamina=player.GetComponent<TheLostShrine.Player.PlayerStamina>();
    player.transform.position=new Vector3(100,100,0);player.GetComponent<Rigidbody2D>().position=player.transform.position;
    target.transform.position=player.transform.position+(Vector3)aim*1.5f;target.GetComponent<Rigidbody2D>().position=target.transform.position;
    Physics2D.SyncTransforms();
    var art=AssetDatabase.LoadAssetAtPath<TheLostShrine.Player.RegisteredActionSprites>(TheLostShrine.Editor.RegisteredSpriteProofBuilder.ForehandAssetPath);
    var settings=weapon.Settings;
    System.Action<float> tick=dt=>weapon.GetType().GetMethod("Simulate",flags).Invoke(weapon,new object[]{dt});
    var presenter=player.GetComponent<TheLostShrine.Player.RegisteredPlayerAnimation>();
    var animation=player.GetComponent<TheLostShrine.Player.PlayerLocomotionAnimation>();
    var view=weapon.GetComponent<TheLostShrine.Weapons.HatchetView>();
    var body=player.transform.Find("Body").GetComponent<SpriteRenderer>();
    var held=body.transform.Find("Drawn held axe").GetComponent<SpriteRenderer>();
    var blade=weapon.transform.Find("Model/Sprite").GetComponent<SpriteRenderer>();
    System.Action draw=()=>{animation.GetType().GetMethod("Present",flags).Invoke(animation,new object[]{.01f});view.GetType().GetMethod("LateUpdate",flags).Invoke(view,null);};
    System.Func<int> cel=()=>{draw();return presenter.DisplayedCel;};
    System.Action reset=()=>{weapon.CancelAction();stamina.Restore();hp.RestoreHealth();weapon.ApplyUpgrades(new TheLostShrine.Weapons.HatchetUpgrade[0]);};
    check(player.TryEquip(weapon),"Real Tutorial prefabs equip");
    reset();target.SetActive(false);weapon.TryLightChop(aim);
    var seen=new System.Collections.Generic.HashSet<int>();bool paired=true;
    for(int i=0;i<30;i++) {
        draw();int frame=presenter.DisplayedCel;seen.Add(frame);
        paired &= presenter.DisplayedAction=="Forehand" && frame>=0 && body.sprite==art.views[direction].body[frame] && held.enabled && held.sprite==art.views[direction].weapon[frame] && !blade.enabled && !body.flipX;
        tick(.01f);
    }
    check(paired && seen.Count==6,"All six actual opening-strike cels pair body/axe without mirroring or duplicate prop");
    check(Mathf.Abs(stamina.Current-(100-settings.lightStaminaCost))<.001f && !weapon.IsImpactPaused,"Miss spends the existing cost once and never gains hit pause");
    weapon.CancelAction();draw();check(!presenter.IsPresentingBody && !held.enabled && blade.enabled,"Cancellation restores ordinary carrying in the same frame");
    target.SetActive(true);Physics2D.SyncTransforms();
    reset();check(weapon.TryLightChop(aim),"Opening forehand begins");
    int before=hp.Health; float baseDuration=weapon.LightDuration;
    tick(baseDuration*settings.lightWindupFraction-.001f);
    check(cel()==2 && hp.Health==before,"Drawn drive precedes real damage");
    tick(.0011f);
    check(cel()==3 && hp.Health==before-settings.lightDamage && weapon.IsImpactPaused,"Real hit selects contact and starts local hit pause");
    float progress=weapon.AttackProgress;tick(.02f);
    check(Mathf.Abs(weapon.AttackProgress-progress)<.000001f && cel()==3,"Actual paused weapon clock holds the contact cel");
    tick(.03f);tick(.07f);
    check(hp.Health==before-settings.lightDamage,"Repeated contact samples cannot duplicate damage");
    tick(.4f);check(weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Presentation sampling does not prolong recovery");

    reset(); upgrade.lightSpeedMultiplier=1.5f;weapon.ApplyUpgrades(new[]{upgrade});
    check(weapon.TryLightChop(aim) && Mathf.Abs(weapon.LightDuration-baseDuration/1.5f)<.00001f,"Speed upgrade changes authoritative action duration");
    tick(weapon.LightDuration*settings.lightWindupFraction+.0001f);
    check(cel()==3,"Upgraded action still selects contact at its damage boundary");
    float durationBeforeSampling=weapon.LightDuration;
    for(int i=0;i<200;i++)cel();
    check(Mathf.Abs(durationBeforeSampling-weapon.LightDuration)<.000001f,"Repeated sprite sampling cannot mutate weapon timing");

    reset();weapon.TryLightChop(aim);before=hp.Health;tick(.2f);
    check(hp.Health==before-settings.lightDamage && weapon.IsImpactPaused,"Long physics step crosses damage once and reports impact pause");
    check(cel()==3 && art.SampleMeleeCel(weapon.AttackProgress,settings.lightWindupFraction,settings.lightSwingEndFraction)>=4,
        "Late-frame hit displays contact rather than freezing follow-through");
    weapon.CancelAction();
    check(weapon.State==TheLostShrine.Weapons.HatchetState.Held && !weapon.IsImpactPaused,"Cancellation clears the action and contact hold");
}
finally {
    Physics2D.simulationMode=oldSimulation;
    foreach(var go in created) if(go!=null)UnityEngine.Object.DestroyImmediate(go);
    UnityEngine.Object.DestroyImmediate(upgrade);
}
return string.Join("\n",results.ToArray());
