// Empty unsaved Play Mode scene; real Tutorial prefabs and actual weapon simulation, no saves.
if (!Application.isPlaying || UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Progression.CheckpointSession>() != null)
    throw new Exception("Use an empty Play Mode scene without a checkpoint session.");
var results=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,label)=>results.Add((ok?"PASS ":"FAIL ")+label);
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var created=new System.Collections.Generic.List<GameObject>();
var oldSimulation=Physics2D.simulationMode;
System.Func<string,GameObject> spawn=path=>{var go=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>(path));created.Add(go);return go;};
try {
    Physics2D.simulationMode=SimulationMode2D.Script;
    var player=spawn("Assets/Prefabs/Player/TutorialPlayer.prefab").GetComponent<TheLostShrine.Player.PlayerCombatController>();
    var weapon=spawn("Assets/Prefabs/Weapons/TutorialHatchet.prefab").GetComponent<TheLostShrine.Weapons.HatchetWeapon>();
    var presenter=player.GetComponent<TheLostShrine.Player.RegisteredPlayerAnimation>();
    var locomotion=player.GetComponent<TheLostShrine.Player.PlayerLocomotionAnimation>();
    var motor=player.GetComponent<Rigidbody2D>();
    var stamina=player.GetComponent<TheLostShrine.Player.PlayerStamina>();
    var view=weapon.GetComponent<TheLostShrine.Weapons.HatchetView>();
    var blade=(SpriteRenderer)view.GetType().GetField("blade",flags).GetValue(view);
    var model=(Transform)view.GetType().GetField("model",flags).GetValue(view);
    var body=player.GetComponentInChildren<Animator>().GetComponent<SpriteRenderer>();
    var held=body.transform.Find("Drawn held axe").GetComponent<SpriteRenderer>();
    var forehand=AssetDatabase.LoadAssetAtPath<TheLostShrine.Player.RegisteredActionSprites>(TheLostShrine.Editor.RegisteredSpriteProofBuilder.ForehandAssetPath);
    var throwArt=AssetDatabase.LoadAssetAtPath<TheLostShrine.Player.RegisteredActionSprites>(TheLostShrine.Editor.RegisteredSpriteProofBuilder.ThrowAssetPath);
    var catchArt=AssetDatabase.LoadAssetAtPath<TheLostShrine.Player.RegisteredActionSprites>(TheLostShrine.Editor.RegisteredSpriteProofBuilder.CatchAssetPath);
    check(catchArt.views.Length==4 && catchArt.views[3].direction==Vector2.left,"All cardinal catches include an authored west view");
    System.Action<float> tick=dt=>weapon.GetType().GetMethod("Simulate",flags).Invoke(weapon,new object[]{dt});
    System.Action<float,float> render=(dt,distance)=>{
        presenter.PrepareFrame(dt,distance);presenter.TryApplyBody();
        view.GetType().GetMethod("LateUpdate",flags).Invoke(view,null);
    };
    System.Action locomote=()=>{locomotion.GetType().GetMethod("LateUpdate",flags).Invoke(locomotion,null);view.GetType().GetMethod("LateUpdate",flags).Invoke(view,null);};
    System.Action reset=()=>{weapon.CancelAction();stamina.Restore();motor.linearVelocity=Vector2.zero;render(.02f,0);locomote();};
    player.transform.position=new Vector3(100,100,0);motor.position=player.transform.position;
    check(presenter!=null && presenter.enabled,"Tutorial prefab has the configured live presenter");
    check(player.TryEquip(weapon),"Tutorial weapon equips");player.UnlockRecall();locomote();
    var original=blade.sprite;var originalScale=blade.transform.localScale;
    check(blade.enabled && !held.enabled,"Ordinary carrying has exactly the original weapon");
    for(int direction=0;direction<4;direction++) {
        Vector2 aim=new[]{Vector2.right,Vector2.up,Vector2.down,Vector2.left}[direction];
        string label=new[]{"East: ","North: ","South: ","West: "}[direction];
        reset();weapon.TryLightChop(aim);render(.02f,0);
        check(presenter.DisplayedAction=="Forehand" && body.sprite==forehand.views[direction].body[0],label+"opening strike selects the full-body drawing");
        check(held.enabled && held.sprite==forehand.views[direction].weapon[0] && !blade.enabled,label+"matching held overlay suppresses the old blade");
        tick(weapon.LightDuration*weapon.Settings.lightWindupFraction+.001f);render(.02f,0);
        check(presenter.DisplayedCel==forehand.contactCel,label+"contact drawing follows gameplay contact");
        weapon.PauseOnImpact(.045f);tick(.02f);render(.02f,0);
        check(presenter.DisplayedCel==forehand.contactCel,label+"hit pause holds contact");
        tick(.5f);weapon.TryLightChop(aim);locomote();
        check(weapon.ComboIndex==1 && !presenter.IsPresentingBody && blade.enabled && !held.enabled,label+"unfinished follow-up retains working combo presentation");
        reset();weapon.TryBeginThrow(aim);render(.02f,0);
        check(presenter.DisplayedAction=="Throw" && presenter.DisplayedCel==0 && !blade.enabled,label+"throw preparation owns body and held axe");
        tick(.08f);render(.02f,0);
        check(presenter.DisplayedCel==2 && weapon.State==TheLostShrine.Weapons.HatchetState.Held,label+"aim loop never releases itself");
        motor.linearVelocity=Vector2.right;
        presenter.PrepareTravelFrame(.02f,Vector2.right*.05f,false);presenter.TryApplyBody();view.GetType().GetMethod("LateUpdate",flags).Invoke(view,null);
        check(!presenter.IsPresentingBody && blade.enabled && !held.enabled,
            label+"deferred moving aim retains the established locomotion and carry fallback");
        motor.linearVelocity=Vector2.zero;render(.02f,0);
        check(presenter.DisplayedAction=="Throw" && !blade.enabled,label+"stopping resumes authored aim");
        weapon.TryReleaseThrow(aim);tick(.039f);render(.02f,0);
        check(!weapon.IsAway && held.enabled && !blade.enabled,label+"axe remains attached before the actual marker");
        tick(.0011f);var physicsOrigin=weapon.transform.position;render(.02f,0);
        check(weapon.IsAway && !held.enabled && blade.enabled && blade.sprite==original,label+"release hands off to the same single hatchet sprite");
        check(weapon.transform.position==physicsOrigin && Vector3.Distance(blade.bounds.center,physicsOrigin)<.0001f,label+"detached sprite spins at the actual collision center");
        check(Mathf.Abs(blade.transform.localScale.x-1.15f)<.0001f,label+"detached size is consistent in every direction");
        check(!blade.flipX && !body.flipX,label+"flight never swaps blade perspective or mirrors the body");
        tick(.2f);render(.02f,0);
        float phase=weapon.FlightSeconds;weapon.TryRecall();
        check(Mathf.Abs(weapon.FlightSeconds-phase)<.00001f,label+"Recall does not restart spin time");
        tick(.005f);render(.02f,0);
        check(weapon.State==TheLostShrine.Weapons.HatchetState.Returning && blade.enabled && !held.enabled,label+"return keeps the complete world prop visible");
        bool hasCatch=direction<catchArt.views.Length;
        check(hasCatch && presenter.DisplayedAction=="Catch" && presenter.DisplayedCel==0 && body.sprite==catchArt.views[direction].body[0]
            && !body.flipX,label+"return uses its authored reach without mirroring the body");
        tick(1);render(.02f,0);
        if(!hasCatch)locomote();
        check(weapon.State==TheLostShrine.Weapons.HatchetState.Held && (hasCatch ? presenter.DisplayedAction=="Catch" && presenter.DisplayedCel==1 : !presenter.IsPresentingBody),label+"confirmed arrival uses the authored grip or ordinary carry");
        check(hasCatch ? !blade.enabled && held.enabled : blade.enabled && !held.enabled,label+"arrival leaves exactly one held axe");
        if(hasCatch)check(body.sprite==catchArt.views[direction].body[1] && held.sprite==catchArt.views[direction].weapon[1],label+"grip uses its matching body and tool cels");
        check(weapon.TryLightChop(aim),label+"catch art adds no attack lock");render(.02f,0);
        check(presenter.DisplayedAction=="Forehand" && held.enabled && !blade.enabled,label+"immediate attack replaces catch cleanly");
        reset();check(blade.enabled && blade.sprite==original && blade.transform.localScale==originalScale && !blade.flipX && !held.enabled,label+"cancellation restores ordinary carry and scale");
    }
    for(int direction=0;direction<4;direction++) {
        reset();Vector2 aim=new[]{Vector2.right,Vector2.up,Vector2.down,Vector2.left}[direction];
        weapon.TryThrow(aim);tick(.1201f);render(.02f,0);
        float previous=Vector2.Dot(model.position,aim);bool forward=true;
        for(int i=0;i<16;i++) {tick(.02f);render(.02f,0);float next=Vector2.Dot(model.position,aim);forward &= next>=previous-.0001f;previous=next;}
        check(forward,new[]{"East","North","South","West"}[direction]+" centered outbound flight never pulls the axe backwards");
    }
    // The same endpoint and pose must survive different simulation step sizes,
    // diagonals, impact distances, waiting on the ground, and a subsequent Recall.
    check(weapon.Settings.throwSpeed==12 && weapon.Settings.throwRange==7,"Tutorial baseline is 12 units/sec over 7 units");
    for(int direction=0;direction<8;direction++) {
        float a=direction*Mathf.PI/4;Vector2 aim=new Vector2(Mathf.Cos(a),Mathf.Sin(a));
        Vector3 endpoint=Vector3.zero;Quaternion landing=Quaternion.identity;
        foreach(float step in new[]{.01f,.037f}) {
            reset();Vector3 start=player.transform.position;
            weapon.TryThrow(aim);
            for(int i=0;i<Mathf.CeilToInt(1.2f/step);i++) {tick(step);render(step,0);}
            check(weapon.State==TheLostShrine.Weapons.HatchetState.Stuck &&
                Vector3.Distance(weapon.transform.position,start+(Vector3)aim*7)<.002f,
                "Direction "+direction+" lands at the exact range with step "+step);
            check(blade.sprite==original && !blade.flipX &&
                Vector3.Distance(blade.bounds.center,weapon.transform.position)<.0001f,
                "Direction "+direction+" ground sprite retains flight art and collision center");
            if(step==.01f) {endpoint=model.position;landing=model.rotation;}
            else check(Vector3.Distance(model.position,endpoint)<.002f && Quaternion.Angle(model.rotation,landing)<.01f,
                "Direction "+direction+" landing pose is independent of final spin phase");
            tick(.25f);render(.25f,0);
            check(Quaternion.Angle(model.rotation,landing)<.01f,"Grounded axe stays planted");
        }
    }
    reset();weapon.TryThrow(Vector2.right);tick(.22f);render(.02f,0);
    float spinAngle=model.eulerAngles.z;
    tick(.02f);render(.02f,0);
    check(Mathf.Abs(Mathf.DeltaAngle(spinAngle,model.eulerAngles.z)+50.4f)<.01f,"Single-sprite spin runs at seven rotations/sec");
    var trail=(TrailRenderer)view.GetType().GetField("trail",flags).GetValue(view);
    check(trail!=null && trail.emitting,"Detached motion uses the existing trail");
    reset();
    var flightWall=new GameObject("Flight stop fixture");created.Add(flightWall);
    var flightWallCollider=flightWall.AddComponent<BoxCollider2D>();flightWallCollider.size=new Vector2(.2f,4);
    Quaternion flightWallLanding=Quaternion.identity;
    foreach(float flightWallDistance in new[]{2f,3.3f}) {
        reset();flightWall.transform.position=player.transform.position+Vector3.right*flightWallDistance;Physics2D.SyncTransforms();
        weapon.TryThrow(Vector2.right);tick(1);render(.02f,0);
        float stopped=weapon.transform.position.x-player.transform.position.x;
        check(weapon.State==TheLostShrine.Weapons.HatchetState.Stuck &&
            Mathf.Abs(stopped-(flightWallDistance-.1f-weapon.Settings.flightRadius))<.01f,
            "Wall stop uses swept contact, not overshoot or bounce");
        check(Vector3.Distance(blade.bounds.center,weapon.transform.position)<.0001f && !trail.emitting,
            "Impact has no height snap and stops the trail");
        if(flightWallDistance==2f)flightWallLanding=model.rotation;
        else check(Quaternion.Angle(model.rotation,flightWallLanding)<.01f,"Impact distance does not change planted orientation");
    }
    flightWall.SetActive(false);reset();
    // Both newer directions expose all eight cels with exclusive weapon ownership.
    foreach(int direction in new[]{2,3}) {
        Vector2 aim=direction==2?Vector2.down:Vector2.left;string label=direction==2?"South":"West";
        reset();weapon.TryBeginThrow(aim);
        var seen=new System.Collections.Generic.HashSet<int>();bool owned=true;
        for(int i=0;i<46;i++) {
            if(i==32)weapon.TryReleaseThrow(aim);
            render(.02f,0);
            if(presenter.DisplayedAction=="Throw") {
                seen.Add(presenter.DisplayedCel);
                owned &= body.sprite==throwArt.views[direction].body[presenter.DisplayedCel] && !body.flipX;
                owned &= weapon.IsAway ? blade.enabled && !held.enabled : !blade.enabled && held.enabled;
                if(!weapon.IsAway)owned &= held.sprite==throwArt.views[direction].weapon[presenter.DisplayedCel];
            }
            tick(.02f);
        }
        check(seen.Count==8,label+" held throw exposes preparation, both aim cels, drive, release and both recovery cels");
        check(owned,label+" uses matching body/tool cels and one complete axe throughout the sequence");
    }
    foreach(int direction in new[]{2,3}) {
        Vector2 aim=direction==2?Vector2.down:Vector2.left;string label=direction==2?"South":"West";
        reset();weapon.TryThrow(aim);tick(.4f);render(.02f,0);weapon.TryRecall();
        tick(.01f);render(.02f,0);tick(1);render(.02f,0);
        render(.05f,0);
        check(presenter.DisplayedAction=="Catch" && presenter.DisplayedCel==2 && body.sprite==catchArt.views[direction].body[2],label+" absorption follows confirmed grip");
        render(.08f,0);
        check(presenter.DisplayedCel==3 && held.sprite==catchArt.views[direction].weapon[3],label+" settles with the same owned axe");
        render(.1f,0);locomote();
        check(!presenter.IsPresentingBody && blade.enabled && !held.enabled,label+" catch returns to original carry without another action lock");
        reset();weapon.TryThrow(aim);tick(.4f);render(.02f,0);weapon.TryRecall();
        motor.linearVelocity=Vector2.right;render(.02f,.02f);tick(1);render(.02f,.02f);locomote();
        check(weapon.State==TheLostShrine.Weapons.HatchetState.Held && !presenter.IsPresentingBody && blade.enabled && !held.enabled,label+" moving catch retains the deferred movement fallback");
    }
    reset();weapon.TryBeginThrow(Vector2.right);tick(.09f);render(.02f,0);
    player.enabled=false;render(.02f,0);
    check(!presenter.IsPresentingBody && !held.enabled,"Control loss clears authored aim and held overlay");player.enabled=true;
    player.GetType().GetMethod("ProcessInput",flags).Invoke(player,new object[]{new TheLostShrine.Input.CombatInputFrame {Active=true}});
    reset();weapon.TryThrow(Vector2.right);tick(.2f);render(.02f,0);
    check(Mathf.Abs(weapon.FlightSeconds-.08f)<.00001f,"Long release step counts only post-marker transit for spin");
    weapon.TryRecall();tick(.01f);render(.02f,0);tick(1);render(.02f,0);
    check(presenter.DisplayedAction=="Catch" && presenter.DisplayedCel==1,"Recall arriving during throw recovery cannot replay a detached throw cel");
    render(.05f,0);check(presenter.DisplayedCel==2,"Catch proceeds into weight absorption");
    render(.18f,0);locomote();check(!presenter.IsPresentingBody && blade.enabled && !held.enabled,"Catch settle returns to locomotion without stale body layers");
    reset();weapon.TryThrow(Vector2.right);tick(1);render(.02f,0);
    check(weapon.State==TheLostShrine.Weapons.HatchetState.Stuck && model.position==weapon.transform.position,"Stopped axe is displayed at its collision landmark");
    weapon.TryRecall();tick(.01f);render(.02f,0);weapon.CancelAction();render(.02f,0);locomote();
    check(presenter.DisplayedAction!="Catch" && blade.enabled && !held.enabled,"Cancelled return cannot leave a waiting reach pose");
    reset();weapon.TryThrow(Vector2.up);tick(.4f);render(.02f,0);weapon.TryRecall();
    motor.linearVelocity=Vector2.right;render(.02f,.02f);tick(1);render(.02f,.02f);locomote();
    check(weapon.State==TheLostShrine.Weapons.HatchetState.Held && !presenter.IsPresentingBody && blade.enabled && !held.enabled,"Moving catch preserves movement and authoritative possession");
    reset();weapon.TryBeginThrow(Vector2.right);tick(.09f);render(.02f,0);weapon.CancelThrow();locomote();
    check(!presenter.IsPresentingBody && !held.enabled && blade.enabled,"Cancelled held aim restores locomotion and carry");
    reset();weapon.TryThrow(Vector2.right);tick(.2f);render(.02f,0);presenter.enabled=false;locomote();
    check(!held.enabled && blade.enabled && blade.sprite==original && blade.transform.localScale==originalScale,"Disabling the presenter restores legacy flight without stale layers");presenter.enabled=true;
    foreach(Vector2 aim in new[]{Vector2.right,Vector2.left}) {
        reset();var wall=new GameObject("Close wall");created.Add(wall);
        wall.transform.position=player.transform.position+(Vector3)aim*.65f;
        wall.AddComponent<BoxCollider2D>().size=new Vector2(.1f,2f);Physics2D.SyncTransforms();
        weapon.TryThrow(aim);tick(.2f);render(.02f,0);
        check(weapon.State==TheLostShrine.Weapons.HatchetState.Stuck && Vector2.Dot(weapon.transform.position,aim)<Vector2.Dot(wall.transform.position,aim) && model.position==weapon.transform.position,
            (aim==Vector2.left?"West":"East")+" close-wall release neither spawns nor draws the stopped prop beyond the wall");
        UnityEngine.Object.DestroyImmediate(wall);
    }
}
finally {
    Physics2D.simulationMode=oldSimulation;
    foreach(var go in created)if(go!=null)UnityEngine.Object.DestroyImmediate(go);
}
return string.Join("\n",results.ToArray());
