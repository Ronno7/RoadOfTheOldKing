// Empty unsaved Play Mode scene. Run once per direction with real prefabs/motor, no save session.
int direction = 0; // 0 east, 1 north, 2 south, 3 west
Vector2 aim=new[]{Vector2.right,Vector2.up,Vector2.down,Vector2.left}[direction];
string facing=new[]{"East","North","South","West"}[direction];
var moveKey=new[]{UnityEngine.InputSystem.Key.D,UnityEngine.InputSystem.Key.W,UnityEngine.InputSystem.Key.S,UnityEngine.InputSystem.Key.A}[direction];
if(!Application.isPlaying || UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Progression.CheckpointSession>()!=null)
    throw new Exception("Use an empty Play Mode scene without a save session.");
var results=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,name)=>results.Add((ok?"PASS ":"FAIL ")+name);
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var root=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Player/TutorialPlayer.prefab"));
var prop=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Weapons/TutorialHatchet.prefab"));
var combat=root.GetComponent<TheLostShrine.Player.PlayerCombatController>();
var weapon=prop.GetComponent<TheLostShrine.Weapons.HatchetWeapon>();
var movement=root.GetComponent<TheLostShrine.Player.PlayerMovement>();
var motor=root.GetComponent<Rigidbody2D>();motor.interpolation=RigidbodyInterpolation2D.None;
var dash=root.GetComponent<TheLostShrine.Player.PlayerDash>();
var stamina=root.GetComponent<TheLostShrine.Player.PlayerStamina>();
var life=root.GetComponent<TheLostShrine.Player.PlayerHealth>();
var presenter=root.GetComponent<TheLostShrine.Player.RegisteredPlayerAnimation>();
var animation=root.GetComponent<TheLostShrine.Player.PlayerLocomotionAnimation>();
var input=root.GetComponent<TheLostShrine.Input.PlayerMovementInput>();
var view=weapon.GetComponent<TheLostShrine.Weapons.HatchetView>();
var body=root.transform.Find("Body").GetComponent<SpriteRenderer>();
var held=body.transform.Find("Drawn held axe").GetComponent<SpriteRenderer>();
var blade=prop.transform.Find("Model/Sprite").GetComponent<SpriteRenderer>();
var art=AssetDatabase.LoadAssetAtPath<TheLostShrine.Player.RegisteredActionSprites>(TheLostShrine.Editor.RegisteredSpriteProofBuilder.SprintAssetPath);
var oldSim=Physics2D.simulationMode;
var oldRouting=UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode;
var keyboard=UnityEngine.InputSystem.InputSystem.AddDevice<UnityEngine.InputSystem.Keyboard>();
var actions=(UnityEngine.InputSystem.InputActionMap)input.GetType().GetField("actions",flags).GetValue(input);
actions.devices=new UnityEngine.InputSystem.InputDevice[]{keyboard};
GameObject wall=null;
System.Action draw=()=>{animation.GetType().GetMethod("Present",flags).Invoke(animation,new object[]{.02f});view.GetType().GetMethod("LateUpdate",flags).Invoke(view,null);};
System.Action<UnityEngine.InputSystem.Key[]> keys=heldKeys=>{
    UnityEngine.InputSystem.InputSystem.QueueStateEvent(keyboard,new UnityEngine.InputSystem.LowLevel.KeyboardState(heldKeys));
    UnityEngine.InputSystem.InputSystem.Update();input.GetType().GetField("hasFocus",flags).SetValue(input,true);
    input.GetType().GetMethod("Update",flags).Invoke(input,null);
};
System.Action step=()=>{movement.GetType().GetMethod("FixedUpdate",flags).Invoke(movement,null);Physics2D.Simulate(.02f);draw();};
System.Action<int> steps=count=>{for(int i=0;i<count;i++)step();};
System.Action reset=()=>{
    weapon.CancelAction();life.HealAtRest();movement.enabled=true;combat.enabled=true;
    combat.GetType().GetMethod("ProcessInput",flags).Invoke(combat,new object[]{new TheLostShrine.Input.CombatInputFrame{Active=true}});
    keys(new UnityEngine.InputSystem.Key[0]);motor.position=new Vector2(200,200);root.transform.position=motor.position;
    motor.linearVelocity=Vector2.zero;Physics2D.SyncTransforms();draw();
};
System.Action sprint=()=>{keys(new[]{moveKey,UnityEngine.InputSystem.Key.LeftShift});step();};
System.Func<float> phase=()=> (float)animation.GetType().GetField("cycle",flags).GetValue(animation);
System.Action<float> tick=t=>weapon.GetType().GetMethod("Simulate",flags).Invoke(weapon,new object[]{t});
try {
    Physics2D.simulationMode=SimulationMode2D.Script;
    UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode=UnityEngine.InputSystem.InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;
    input.GetType().GetField("paused",flags).SetValue(input,false);
    string error;check(art.IsRegistered(out error) && art.celCount==8 && art.views.Length==4 && art.views[direction].direction==aim,"Eight registered cels per cardinal view share foot pivot and density");
    check(art.contactCel==-1 && art.possessionCel==-1 && art.exposures.Length==0,"Distance loop cannot dispatch combat or possession markers");
    reset();sprint();var origin=new Vector2(200,200);var seen=new System.Collections.Generic.HashSet<int>();bool unarmed=true;
    for(int i=0;i<23;i++){int cel=presenter.DisplayedCel;seen.Add(cel);unarmed&=cel>=0 && presenter.DisplayedAction=="Sprint" && !held.enabled && body.sprite==((direction!=1)?art.views[direction].unarmedBody[cel]:art.views[direction].body[cel]);step();}
    check(combat.Weapon==null && unarmed && seen.Count==8,"Before pickup, all eight sprint drawings play without a phantom axe");
    check(Mathf.Abs(Vector2.Distance(motor.position,origin)-movement.SprintSpeed*.02f*24)<.003f && Mathf.Abs(motor.linearVelocity.magnitude-7.2f)<.001f,"Authored sprint preserves the existing 7.2 units/second motor speed");
    check(Mathf.Abs(stamina.Current-(100-20*.02f*24))<.002f,"Only the motor spends the existing 20 stamina/second");
    check(Mathf.Abs(phase()-Mathf.Repeat(7.2f*.02f*24/3.2f,1))<.001f,"Eight-cel cycle wraps every 3.2 units of actual travel");
    keys(new UnityEngine.InputSystem.Key[0]);step();
    check(presenter.DisplayedAction=="Locomotion" && body.sprite.name==facing+"_00" && phase()==0 && motor.linearVelocity==Vector2.zero,"Releasing movement immediately restores original idle and resets gait");
    sprint();check(presenter.DisplayedCel==0,"Fresh sprint begins on the contact pose");
    float before=phase();keys(new[]{moveKey});step();
    check(!movement.IsSprinting && body.sprite.name.StartsWith(facing+"_") && !presenter.IsPresentingBody && Mathf.Abs(phase()-Mathf.Repeat(before+4.5f*.02f/2.25f,1))<.001f,"Sprint-to-walk preserves phase and restores the original sprite writer");
    before=phase();sprint();
    check(presenter.DisplayedAction=="Sprint" && Mathf.Abs(phase()-Mathf.Repeat(before+7.2f*.02f/3.2f,1))<.001f,"Walk-to-sprint preserves phase at its authored stride length");
    before=phase();keys(new[]{UnityEngine.InputSystem.Key.A,UnityEngine.InputSystem.Key.LeftShift});step();
    check(movement.IsSprinting && presenter.DisplayedAction=="Sprint" && body.sprite==art.views[3].unarmedBody[presenter.DisplayedCel] && !body.flipX && Mathf.Abs(phase()-Mathf.Repeat(before+7.2f*.02f/3.2f,1))<.001f,"West sprint uses its registered view and shared phase without runtime flipping");
    before=phase();sprint();check(presenter.DisplayedAction=="Sprint" && Mathf.Abs(phase()-Mathf.Repeat(before+7.2f*.02f/3.2f,1))<.001f && !body.flipX,"Turning back to the registered view keeps phase without runtime flipping");
    before=phase();keys(new[]{direction==0?UnityEngine.InputSystem.Key.W:UnityEngine.InputSystem.Key.D,UnityEngine.InputSystem.Key.LeftShift});step();
    check(presenter.DisplayedAction=="Sprint" && body.sprite.name.StartsWith(direction==0?"North_SprintBody":"East_SprintUnarmed") && Mathf.Abs(phase()-Mathf.Repeat(before+7.2f*.02f/3.2f,1))<.001f,"Authored sprint turns preserve phase and select the correct authored view");
    sprint();
    for(int i=0;i<15;i++)draw();
    check(!presenter.IsPresentingBody && body.sprite.name==facing+"_00" && phase()==0,"Repeated render samples without travel cannot run the feet in place");

    reset();sprint();float equipPhase=phase();check(combat.TryEquip(weapon),"Real Tutorial hatchet equips");step();
    check(Mathf.Abs(phase()-Mathf.Repeat(equipPhase+7.2f*.02f/3.2f,1))<.001f && body.sprite==art.views[direction].body[presenter.DisplayedCel],"Equipping during sprint selects the carrying pose without restarting the stride");
    seen.Clear();bool paired=true;
    for(int i=0;i<23;i++){int cel=presenter.DisplayedCel;seen.Add(cel);paired&=cel>=0 && body.sprite==art.views[direction].body[cel] && held.enabled && held.sprite==art.views[direction].weapon[cel] && !blade.enabled;step();}
    check(paired && seen.Count==8,"Equipped sprint pairs all eight body/axe cels with exactly one visible weapon");
    keys(new[]{moveKey});step();check(!held.enabled && blade.enabled && !presenter.IsPresentingBody,"Walk handoff restores the ordinary carry prop on the same frame");
    reset();sprint();check(weapon.TryLightChop(aim),"Sprint permits the existing attack transition");draw();
    check(presenter.DisplayedAction=="Forehand" && held.enabled && !blade.enabled,"Same-frame attack overrides sprint with its authored cardinal view");
    reset();sprint();check(weapon.TryBeginThrow(aim),"Sprint can enter held-E preparation");tick(.09f);step();
    check(!movement.IsSprinting && presenter.DisplayedAction!="Sprint" && Mathf.Abs(motor.linearVelocity.magnitude-2.475f)<.001f,"Moving aim suppresses sprint and retains its existing 55-percent walk speed");
    weapon.CancelThrow();step();check(movement.IsSprinting && presenter.DisplayedAction=="Sprint","Canceling aim restores sprint on the next motor step");
    reset();sprint();keys(new[]{moveKey,UnityEngine.InputSystem.Key.LeftShift,UnityEngine.InputSystem.Key.Space});step();
    check(dash.IsDashing && presenter.DisplayedAction=="Dash" && held.enabled,"Accepted dash immediately owns the presentation");
    keys(new[]{moveKey,UnityEngine.InputSystem.Key.LeftShift});steps(11);
    check(!dash.IsDashing && movement.IsSprinting && presenter.DisplayedAction=="Sprint","Held sprint resumes immediately after dash travel without recovery lock");

    reset();wall=new GameObject("Sprint verification wall");wall.layer=LayerMask.NameToLayer("Environment");wall.transform.position=new Vector3(200,200,0)+(Vector3)aim*1.5f;wall.AddComponent<BoxCollider2D>().size=Mathf.Abs(aim.x)>.5f?new Vector2(1,8):new Vector2(8,1);Physics2D.SyncTransforms();sprint();steps(20);
    check(Vector2.Dot(motor.position-new Vector2(200,200),aim)<.64f && !presenter.IsPresentingBody && body.sprite.name==facing+"_00","Collision stops the drawn feet at the wall");
    UnityEngine.Object.DestroyImmediate(wall);wall=null;
    reset();sprint();stamina.TrySpend(stamina.Current);step();
    check(!movement.IsSprinting && !presenter.IsPresentingBody && Mathf.Abs(motor.linearVelocity.magnitude-4.5f)<.001f,"Exhaustion immediately restores walk speed and art");
    reset();sprint();movement.enabled=false;draw();check(!presenter.IsPresentingBody && !held.enabled && body.sprite.name==facing+"_00","Disabling the motor clears sprint and held layers");
    reset();sprint();combat.enabled=false;draw();check(!presenter.IsPresentingBody && !held.enabled,"Control loss clears sprint art");
    reset();sprint();motor.position+=Vector2.right*10;root.transform.position=motor.position;draw();
    check(!presenter.IsPresentingBody && phase()==0,"Teleport resets the old sprint instead of fast-forwarding it");
    reset();sprint();input.GetType().GetMethod("OnApplicationFocus",flags).Invoke(input,new object[]{false});step();
    check(!movement.IsSprinting && !presenter.IsPresentingBody && motor.linearVelocity==Vector2.zero,"Focus loss stops sprint movement and presentation");
    reset();sprint();motor.simulated=false;draw();check(!presenter.IsPresentingBody,"A disabled physics body cannot continue displaying sprint");motor.simulated=true;

    reset();weapon.TryThrow(aim);tick(.4f);draw();sprint();
    seen.Clear();bool awayPoses=true;
    for(int i=0;i<23;i++) {
        int cel=presenter.DisplayedCel;seen.Add(cel);
        awayPoses &= cel>=0 && body.sprite==((direction!=1)?art.views[direction].unarmedBody[cel]:art.views[direction].body[cel]) && !held.enabled && blade.enabled;
        step();
    }
    check(weapon.IsAway && presenter.DisplayedAction=="Sprint" && awayPoses && seen.Count==8,"All eight axe-away sprint cels select empty-hand poses while the complete world prop remains visible");
    combat.UnlockRecall();check(weapon.TryRecall(),"Real Recall accepts the moving owner");bool arrived=false,ownership=true;
    for(int i=0;i<100 && !arrived;i++){step();tick(.02f);view.GetType().GetMethod("LateUpdate",flags).Invoke(view,null);arrived=weapon.State==TheLostShrine.Weapons.HatchetState.Held;if(!arrived)ownership&=!held.enabled && blade.enabled;}
    float catchPhase=phase();step();check(arrived && ownership && presenter.DisplayedAction=="Sprint" && held.enabled && !blade.enabled && body.sprite==art.views[direction].body[presenter.DisplayedCel] && Mathf.Abs(phase()-Mathf.Repeat(catchPhase+7.2f*.02f/3.2f,1))<.001f,"Recall arrives during sprint and restores one held axe without a stationary catch lock");
    check(body.transform.localScale==Vector3.one && body.transform.localPosition==new Vector3(0,-.25f,0) && !body.flipX,"Sprint preserves body scale and registration without runtime flipping");
}
finally {
    if(wall!=null)UnityEngine.Object.DestroyImmediate(wall);
    Physics2D.simulationMode=oldSim;
    UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode=oldRouting;
    UnityEngine.InputSystem.InputSystem.RemoveDevice(keyboard);
    UnityEngine.Object.DestroyImmediate(root);UnityEngine.Object.DestroyImmediate(prop);
}
return new{passed=results.Count(s=>s.StartsWith("PASS")),failed=results.Count(s=>s.StartsWith("FAIL")),results};
