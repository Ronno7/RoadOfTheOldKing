// Tutorial Play Mode. Uses an isolated target at (200,200); no save data is touched.
// Stop Play Mode afterward to restore the stump pickup.
if (!Application.isPlaying) throw new Exception("Enter Tutorial Play Mode first.");
var p = GameObject.Find("Player");
var combat = p.GetComponent<TheLostShrine.Player.PlayerCombatController>();
var weapon = GameObject.Find("Hatchet Pickup").GetComponent<TheLostShrine.Weapons.HatchetWeapon>();
if (combat.Weapon == null) combat.TryEquip(weapon);
var animation = p.GetComponent<TheLostShrine.Player.PlayerLocomotionAnimation>();
var chop = p.GetComponent<TheLostShrine.Player.PlayerChopAnimation>();
var movement = p.GetComponent<TheLostShrine.Player.PlayerMovement>();
var input = p.GetComponent<TheLostShrine.Input.PlayerMovementInput>();
var body = p.GetComponent<Rigidbody2D>();
var health = p.GetComponent<TheLostShrine.Player.PlayerHealth>();
var sprite = p.transform.Find("Body").GetComponent<SpriteRenderer>();
var view = weapon.GetComponent<TheLostShrine.Weapons.HatchetView>();
var model = weapon.transform.Find("Model");
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic;
var actions = (UnityEngine.InputSystem.InputActionMap)input.GetType().GetField("actions",flags).GetValue(input);
var oldDevices = actions.devices;
var oldRouting = UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode;
var oldFocus = input.GetType().GetField("hasFocus",flags).GetValue(input);
var oldPause = input.GetType().GetField("paused",flags).GetValue(input);
var keyboard = UnityEngine.InputSystem.InputSystem.AddDevice<UnityEngine.InputSystem.Keyboard>();
var oldPosition = body.position;
var oldInterpolation = body.interpolation;
var oldSimulation = Physics2D.simulationMode;
var oldFacing = movement.FacingDirection;
var created = new System.Collections.Generic.List<GameObject>();
var results = new System.Collections.Generic.List<string>();
System.Action<bool,string> check = (ok,label) => results.Add((ok ? "PASS " : "FAIL ") + label);
System.Action<float> tick = dt => { while(dt > .00001f) { float step=Mathf.Min(dt,.02f); weapon.GetType().GetMethod("Simulate",flags).Invoke(weapon,new object[]{step}); dt-=step; } };
System.Action render = () => {
    animation.GetType().GetMethod("LateUpdate",flags).Invoke(animation,null);
    view.GetType().GetMethod("LateUpdate",flags).Invoke(view,null);
};
System.Action<UnityEngine.InputSystem.Key[]> keys = held => {
    UnityEngine.InputSystem.InputSystem.QueueStateEvent(keyboard,new UnityEngine.InputSystem.LowLevel.KeyboardState(held));
    UnityEngine.InputSystem.InputSystem.Update();
    input.GetType().GetField("hasFocus",flags).SetValue(input,true);
    input.GetType().GetField("paused",flags).SetValue(input,false);
    input.GetType().GetMethod("Update",flags).Invoke(input,null);
};
System.Action reset = () => { weapon.CancelAction(); health.HealAtRest(); body.linearVelocity=Vector2.zero; };
int confirmed = 0;
System.Action<TheLostShrine.Combat.CombatHit> onHit = hit => confirmed++;
weapon.HitConfirmed += onHit;
try
{
    Physics2D.simulationMode=SimulationMode2D.Script;
    body.interpolation=RigidbodyInterpolation2D.None;
    body.position=new Vector2(200,200);p.transform.position=body.position;
    UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode=UnityEngine.InputSystem.InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;
    actions.devices=new UnityEngine.InputSystem.InputDevice[]{keyboard};
    keys(new[]{UnityEngine.InputSystem.Key.D});
    var target=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Combat/TutorialPracticeDummy.prefab"));
    created.Add(target);target.transform.position=new Vector3(202,200,0);target.GetComponent<Rigidbody2D>().position=target.transform.position;
    var extra=new GameObject("Second hit collider");extra.transform.SetParent(target.transform,false);extra.AddComponent<CircleCollider2D>().radius=.2f;
    var targetHealth=target.GetComponent<TheLostShrine.Combat.Damageable>();
    Physics2D.SyncTransforms();reset();float stamina=p.GetComponent<TheLostShrine.Player.PlayerStamina>().Current;
    check(weapon.TryLightChop(Vector2.right),"East chop begins from Held");render();
    check(chop.IsPlaying && sprite.sprite.name=="ChopEast_00","Combat owns the anticipation cel");
    check(p.GetComponent<TheLostShrine.Player.PlayerStamina>().Current==stamina-weapon.Settings.lightStaminaCost,"One stamina payment per attack");
    movement.GetType().GetMethod("FixedUpdate",flags).Invoke(movement,null);
    check(body.linearVelocity==Vector2.zero && movement.FacingDirection==Vector2.right,"Movement input plants the feet and locks attack facing");
    check(!p.GetComponent<TheLostShrine.Player.PlayerDash>().TryStart(Vector2.down),"Dodge cannot interrupt committed attack");
    tick(.06f);render();
    check(sprite.sprite.name=="ChopEast_01" && targetHealth.Health==100 && confirmed==0,"Raised windup has no premature damage");
    tick(.04f);render();
    check(sprite.sprite.name=="ChopEast_02" && targetHealth.Health==90 && confirmed==1,"Contact cel and one confirmed hit occur together");
    check(weapon.IsImpactPaused && Time.timeScale==1f,"Impact pauses this action without freezing world time");
    var pose=sprite.sprite;var hand=model.position;var progress=weapon.AttackProgress;
    tick(.02f);render();
    check(weapon.AttackProgress==progress && sprite.sprite==pose && model.position==hand,"Body, grip and blade hold together during impact pause");
    tick(.025f);tick(.04f);render();
    check(!weapon.IsImpactPaused && weapon.AttackProgress>progress && targetHealth.Health==90 && confirmed==1,"Action resumes and multiple colliders cannot duplicate damage");
    tick(.05f);render();
    check(sprite.sprite.name=="ChopEast_03","Follow-through uses a dedicated body cel");
    tick(.06f);movement.GetType().GetMethod("FixedUpdate",flags).Invoke(movement,null);
    check(body.linearVelocity.x>0 && body.linearVelocity.x<movement.MoveSpeed,"Movement eases back during late recovery");
    tick(.07f);render();
    check(weapon.State==TheLostShrine.Weapons.HatchetState.Held && !chop.IsPlaying && !sprite.sprite.name.StartsWith("Chop"),"Completed attack returns body ownership to locomotion");
    health.HealAtRest();
    check(weapon.TryLightChop(Vector2.right) && weapon.ComboIndex==1,"Second combo input remains available");
    tick(.5f);health.HealAtRest();
    check(weapon.TryLightChop(Vector2.right) && weapon.ComboIndex==2,"Third combo input retains its finisher stage");
    tick(.5f);check(targetHealth.Health==65,"Combo retains 10/10/15 damage");
    target.SetActive(false);reset();confirmed=0;
    weapon.TryLightChop(Vector2.right);tick(.10f);
    check(!weapon.IsImpactPaused && confirmed==0,"Missed swings have no impact pause or hit feedback");
    tick(.21f);check(weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Miss duration remains the authored 0.3 seconds");
    reset();weapon.TryLightChop(Vector2.left);render();
    check(!chop.IsPlaying && !sprite.sprite.name.StartsWith("Chop"),"Other directions retain the existing presentation pending new art");
    reset();weapon.TryLightChop(Vector2.right);weapon.PauseOnImpact(.045f);weapon.CancelAction();render();
    check(!weapon.IsImpactPaused && !chop.IsPlaying && weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Cancellation clears impact pause and pose ownership");
    var guarded=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Combat/ShieldPracticeDummy.prefab"));
    created.Add(guarded);guarded.transform.position=new Vector3(201.5f,200,0);guarded.GetComponent<Rigidbody2D>().position=guarded.transform.position;
    Physics2D.SyncTransforms();reset();confirmed=0;weapon.TryLightChop(Vector2.right);tick(.1f);
    check(confirmed==0 && !weapon.IsImpactPaused && guarded.GetComponent<TheLostShrine.Combat.Damageable>().Health==100,"Blocked damage does not emit a successful-hit event");
    guarded.SetActive(false);reset();target.SetActive(true);
    targetHealth.RestoreHealth();target.GetComponent<Collider2D>().enabled=true;
    var wall=new GameObject("Occluding wall");created.Add(wall);wall.transform.position=new Vector3(201,200,0);wall.AddComponent<BoxCollider2D>().size=new Vector2(.2f,2);
    Physics2D.SyncTransforms();confirmed=0;weapon.TryLightChop(Vector2.right);tick(.1f);
    check(targetHealth.Health==100 && confirmed==0,"Solid boundaries still block melee hits");
    reset();var feedback=target.GetComponent<TheLostShrine.Combat.WoodTargetFeedback>();
    var root=target.transform.position;var visual=target.transform.Find("Target Visual");var rest=visual.localPosition;
    targetHealth.ReceiveHit(new TheLostShrine.Combat.CombatHit(p,TheLostShrine.Combat.AttackKind.LightChop,10,Vector2.right,0,.1f));
    feedback.GetType().GetMethod("Tick",flags).Invoke(feedback,new object[]{.02f});
    check(visual.localPosition!=rest && target.transform.position==root,"Target recoil affects artwork, preserving the collision footprint");
    feedback.GetType().GetMethod("Tick",flags).Invoke(feedback,new object[]{.5f});
    check(visual.localPosition==rest && !target.GetComponentsInChildren<SpriteRenderer>().Any(s=>s.name=="Wood chip" && s.enabled),"Recoil settles and pooled chips disappear");
    check(AssetDatabase.LoadAssetAtPath<TheLostShrine.Weapons.HatchetSettings>("Assets/Settings/Weapons/HatchetSettings.asset")!=weapon.Settings,"Tutorial timing is separate from PrototypeLoop settings");
}
finally
{
    weapon.HitConfirmed-=onHit;reset();keys(new UnityEngine.InputSystem.Key[0]);
    actions.devices=oldDevices;UnityEngine.InputSystem.InputSystem.RemoveDevice(keyboard);
    UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode=oldRouting;
    input.GetType().GetField("hasFocus",flags).SetValue(input,oldFocus);
    input.GetType().GetField("paused",flags).SetValue(input,oldPause);
    foreach(var go in created) UnityEngine.Object.DestroyImmediate(go);
    body.position=oldPosition;p.transform.position=oldPosition;body.interpolation=oldInterpolation;
    movement.GetType().GetProperty("FacingDirection").SetValue(movement,oldFacing,null);
    Physics2D.simulationMode=oldSimulation;Physics2D.SyncTransforms();render();
}
return string.Join("\n",results.ToArray());
