// Run through Unity MCP execute_code in a fresh PrototypeLoop Play Mode session
// with an empty TheLostShrine.Verification.* saveKey. Stop Play Mode afterward.
// Uses the real input actions, movement, weapon, stamina and bonfire components.
var results = new System.Collections.Generic.List<string>();
System.Action<bool,string> check = (ok,name) => results.Add((ok ? "PASS " : "FAIL ") + name);
System.Func<float,float,bool> near = (a,b) => Mathf.Abs(a-b) < .02f;
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic;
var session = TheLostShrine.Progression.CheckpointSession.Instance;
if (!((string)session.GetType().GetField("saveKey",flags).GetValue(session)).StartsWith("TheLostShrine.Verification."))
    throw new System.Exception("Use an isolated verification save slot.");
var player = GameObject.Find("Player").GetComponent<TheLostShrine.Player.PlayerCombatController>();
var stamina = player.GetComponent<TheLostShrine.Player.PlayerStamina>();
var life = player.GetComponent<TheLostShrine.Player.PlayerHealth>();
var movement = player.GetComponent<TheLostShrine.Player.PlayerMovement>();
var input = player.GetComponent<TheLostShrine.Input.PlayerMovementInput>();
var body = player.GetComponent<Rigidbody2D>();
var reaction = player.GetComponent<TheLostShrine.Combat.HitReaction>();
var weapon = UnityEngine.Object.FindFirstObjectByType<TheLostShrine.Weapons.HatchetWeapon>();
var sim = weapon.GetType().GetMethod("Simulate",flags);
var recover = stamina.GetType().GetMethod("Tick",flags);
var moveUpdate = movement.GetType().GetMethod("FixedUpdate",flags);
var inputUpdate = input.GetType().GetMethod("Update",flags);
var previousInputRouting = UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode;
UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode = UnityEngine.InputSystem.InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;
var keyboard = UnityEngine.InputSystem.InputSystem.AddDevice<UnityEngine.InputSystem.Keyboard>();
var movementActions = (UnityEngine.InputSystem.InputActionMap)input.GetType().GetField("actions",flags).GetValue(input);
var previousDevices = movementActions.devices;
movementActions.devices = new UnityEngine.InputSystem.InputDevice[]{keyboard};
var oldSimulation = Physics2D.simulationMode;
var oldFocus = (bool)input.GetType().GetField("hasFocus",flags).GetValue(input);
var oldPaused = (bool)input.GetType().GetField("paused",flags).GetValue(input);
var originalPosition = body.position;
GameObject target = null;
System.Action<float> tick = seconds => { for (int i=0;i<Mathf.CeilToInt(seconds/.02f);i++) { sim.Invoke(weapon,new object[]{.02f}); recover.Invoke(stamina,new object[]{.02f}); } };
System.Action<UnityEngine.InputSystem.Key[]> keys = held => {
    UnityEngine.InputSystem.InputSystem.QueueStateEvent(keyboard,new UnityEngine.InputSystem.LowLevel.KeyboardState(held));
    UnityEngine.InputSystem.InputSystem.Update();
    input.GetType().GetField("hasFocus",flags).SetValue(input,true);
    inputUpdate.Invoke(input,null);
};
System.Action moveTick = () => moveUpdate.Invoke(movement,null);
System.Action<Vector2> move = p => { body.position=p; player.transform.position=p; body.linearVelocity=Vector2.zero; Physics2D.SyncTransforms(); };
try {
    Physics2D.simulationMode=SimulationMode2D.Script;
    input.GetType().GetField("paused",flags).SetValue(input,false);
    input.GetType().GetField("hasFocus",flags).SetValue(input,true);
    check(stamina != null && near(stamina.Current,100) && life.Health.Health==100,"Player starts with 100 stamina and HP");
    check(!weapon.TryLightChop(Vector2.right) && !weapon.TryThrow(Vector2.right) && near(stamina.Current,100),"Unowned weapon cannot spend player stamina");
    player.TryEquip(weapon); move(new Vector2(100,100));
    keys(new[]{UnityEngine.InputSystem.Key.W}); moveTick();
    check(near(body.linearVelocity.magnitude,4.5f) && near(stamina.Current,100),"Walking is free at 4.5 units per second");
    keys(new[]{UnityEngine.InputSystem.Key.W,UnityEngine.InputSystem.Key.LeftShift}); moveTick();
    check(input.SprintHeld && movement.IsSprinting && near(body.linearVelocity.magnitude,7.2f),"Left Shift sprints through actual input bindings");
    check(near(stamina.Current,99.6f),"Sprint drains 20 stamina per second");
    keys(new[]{UnityEngine.InputSystem.Key.W,UnityEngine.InputSystem.Key.D,UnityEngine.InputSystem.Key.RightShift}); moveTick();
    check(movement.IsSprinting && near(body.linearVelocity.magnitude,7.2f),"Right Shift works and diagonal sprint has equal speed");
    keys(new[]{UnityEngine.InputSystem.Key.LeftShift}); float idleStamina=stamina.Current; moveTick();
    check(!movement.IsSprinting && body.linearVelocity==Vector2.zero && near(stamina.Current,idleStamina),"Holding Shift while stationary spends nothing");
    keys(new[]{UnityEngine.InputSystem.Key.W,UnityEngine.InputSystem.Key.LeftShift}); input.enabled=false; moveTick();
    check(!input.SprintHeld && !movement.IsSprinting && body.linearVelocity==Vector2.zero,"Disabling movement input clears sprint and stops movement");
    input.enabled=true; input.GetType().GetMethod("OnApplicationFocus",flags).Invoke(input,new object[]{false}); inputUpdate.Invoke(input,null);
    check(!input.SprintHeld && input.MoveDirection==Vector2.zero,"Losing focus clears held sprint and movement");
    input.GetType().GetMethod("OnApplicationFocus",flags).Invoke(input,new object[]{true});
    keys(new UnityEngine.InputSystem.Key[0]);
    keys(new[]{UnityEngine.InputSystem.Key.W,UnityEngine.InputSystem.Key.LeftShift}); stamina.Restore();
    for(int i=0;i<250;i++)moveTick();
    check(stamina.Current<.02f && stamina.Current>=0,"Five seconds of sprint exhausts the pool without going negative");
    moveTick();check(!movement.IsSprinting && near(body.linearVelocity.magnitude,4.5f),"Exhaustion falls back to walking without locking movement");
    recover.Invoke(stamina,new object[]{.8f}); check(stamina.Current<.02f,"Recovery waits through its delay");
    recover.Invoke(stamina,new object[]{.4f}); moveTick();check(!movement.IsSprinting && near(stamina.Current,10),"Sprint cannot jitter back on below its 20 stamina threshold");
    recover.Invoke(stamina,new object[]{.4f});moveTick();check(movement.IsSprinting && near(stamina.Current,19.6f),"Holding Shift resumes sprint after recovering 20 stamina");
    keys(new UnityEngine.InputSystem.Key[0]);moveTick();stamina.Restore();

    target=UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Combat/MeleeSentinel.prefab"));
    target.name="Stamina Verification Sentinel";target.transform.position=body.position+Vector2.right*1.1f;
    target.GetComponent<Rigidbody2D>().position=target.transform.position;Physics2D.SyncTransforms();
    var enemyHealth=target.GetComponent<TheLostShrine.Combat.Damageable>();
    check(enemyHealth.MaxHealth==100,"Sentinel uses the expanded 100 HP pool");
    check(weapon.TryLightChop(Vector2.right) && near(stamina.Current,82),"First slash costs 18 immediately");
    check(!weapon.TryThrow(Vector2.right) && !weapon.TryBeginCharge() && near(stamina.Current,82),"Busy actions reject without charging extra stamina");
    tick(.22f);check(enemyHealth.Health==90,"First slash deals 10 damage");
    check(weapon.TryLightChop(Vector2.right) && near(stamina.Current,64),"Second slash costs 18");tick(.22f);
    check(weapon.TryLightChop(Vector2.right) && near(stamina.Current,40),"Finisher costs 24");tick(.26f);
    check(enemyHealth.Health==65 && near(stamina.Current,40),"Complete combo deals 35 damage, costs 60 and cannot regenerate mid-action");
    weapon.TryLightChop(Vector2.right);tick(.22f);weapon.TryLightChop(Vector2.right);tick(.22f);
    int previousCombo=weapon.ComboIndex;
    check(!weapon.TryLightChop(Vector2.right) && near(stamina.Current,4) && weapon.ComboIndex==previousCombo,"Repeated slashes exhaust the pool; rejected finisher does not advance combo");
    check(enemyHealth.Health==45 && enemyHealth.IsAlive,"A full stamina burst cannot kill the sentinel before recovery matters");
    check(stamina.WasSpendRejected,"Rejected actions expose feedback for the HUD");
    tick(.82f);check(stamina.Current>4,"Stamina recovers once the player stops attacking");
    weapon.CancelAction();target.SetActive(false);stamina.Restore();
    check(weapon.TryBeginCharge() && near(stamina.Current,65),"Cleave pays 35 once at charge start");
    keys(new[]{UnityEngine.InputSystem.Key.W,UnityEngine.InputSystem.Key.LeftShift});moveTick();
    check(!movement.IsSprinting && near(body.linearVelocity.magnitude,4.5f),"Charging permits walking but prevents sprint");
    tick(3f);check(near(stamina.Current,65),"Holding a full charge does not regenerate its stamina cost");
    check(weapon.TryReleaseCharge(Vector2.right) && near(stamina.Current,65),"Cleave release does not charge twice");moveTick();
    check(!movement.IsSprinting,"Cleave animation also prevents sprint");tick(.46f);
    stamina.Restore();weapon.TryBeginCharge();weapon.CancelCharge();
    check(near(stamina.Current,65) && weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Cancelling a charge does not refund its commitment cost");
    stamina.Restore();weapon.TryBeginCharge();tick(.1f);
    check(!weapon.TryReleaseCharge(Vector2.right) && near(stamina.Current,65),"An early charge release cannot create a free attack");
    keys(new UnityEngine.InputSystem.Key[0]);moveTick();weapon.CancelAction();stamina.Restore();
    check(weapon.TryThrow(Vector2.right) && near(stamina.Current,75),"Throw spends 25 once at launch");
    check(!weapon.TryThrow(Vector2.right) && near(stamina.Current,75),"Repeated E while away cannot charge another throw");
    stamina.TrySpend(75);check(!weapon.TryRecall(),"Zero stamina does not bypass the Recall unlock requirement");
    player.UnlockRecall();check(weapon.TryRecall() && near(stamina.Current,0),"Unlocked Recall remains available at zero stamina");tick(.04f);
    check(weapon.State==TheLostShrine.Weapons.HatchetState.Held,"Free Recall retrieves the weapon at exhaustion");
    check(!weapon.TryLightChop(Vector2.right) && !weapon.TryBeginCharge() && !weapon.TryThrow(Vector2.right),"Empty stamina blocks both attacks and new throws");
    check(weapon.State==TheLostShrine.Weapons.HatchetState.Held && near(stamina.Current,0),"Rejected actions leave the weapon usable and stamina unchanged");
    stamina.Restore();weapon.CancelAction();
    for(int i=0;i<4;i++){weapon.TryThrow(Vector2.right);weapon.TryRecall();tick(.04f);}
    check(near(stamina.Current,0) && !weapon.TryThrow(Vector2.right),"Rapid throw/Recall spam stops after four throws");
    recover.Invoke(stamina,new object[]{.4f});weapon.TryThrow(Vector2.right);
    recover.Invoke(stamina,new object[]{.5f});check(stamina.Current>0,"Failed spam inputs do not continually postpone recovery");
    recover.Invoke(stamina,new object[]{20f});check(near(stamina.Current,100),"Recovery clamps at the maximum");
    check(!stamina.TrySpend(-10) && !stamina.TrySpend(float.NaN) && near(stamina.Current,100),"Invalid costs cannot grant or corrupt stamina");

    keys(new[]{UnityEngine.InputSystem.Key.W,UnityEngine.InputSystem.Key.LeftShift});stamina.Restore();moveTick();
    life.Health.ReceiveHit(new TheLostShrine.Combat.CombatHit(target,TheLostShrine.Combat.AttackKind.EnemyMelee,20,Vector2.left,2,.2f));
    float beforeHitTick=stamina.Current;var impulse=body.linearVelocity;moveTick();
    check(!movement.IsSprinting && near(stamina.Current,beforeHitTick) && body.linearVelocity==impulse,"Stagger interrupts sprint without overriding knockback or draining more stamina");
    check(life.Health.Health==80,"An enemy hit removes 20 of the player's 100 HP");reaction.Clear();
    var camp=session.Fires[0];move(camp.SpawnPosition);stamina.TrySpend(40);
    var interaction=player.GetComponent<TheLostShrine.Player.PlayerBonfireInteraction>();
    check(interaction.Open(camp) && near(stamina.Current,100) && life.Health.Health==100,"Bonfire rest restores HP and stamina");moveTick();
    check(!movement.IsSprinting && !input.SprintHeld,"Bonfire menu cancels held sprint");interaction.Close();
    stamina.TrySpend(60);var exit=session.Fires[1];move(exit.SpawnPosition);session.Rest(exit);stamina.TrySpend(40);
    check(session.Travel(camp,exit) && near(stamina.Current,100),"Bonfire travel restores stamina");
    stamina.TrySpend(50);life.GetType().GetField("invulnerableUntil",flags).SetValue(life,0f);
    life.Health.ReceiveHit(new TheLostShrine.Combat.CombatHit(target,TheLostShrine.Combat.AttackKind.EnemyMelee,100,Vector2.down,0,0));
    float deadStamina=stamina.Current;recover.Invoke(stamina,new object[]{10f});
    check(!life.IsAlive && !movement.enabled && !movement.IsSprinting && near(stamina.Current,deadStamina),"Death stops sprint and stamina regeneration");
    check(!weapon.TryThrow(Vector2.right) && !stamina.TrySpend(1),"Dead player cannot spend stamina or launch attacks");
} finally {
    UnityEngine.InputSystem.InputSystem.settings.editorInputBehaviorInPlayMode = previousInputRouting;
    movementActions.devices = previousDevices;
    UnityEngine.InputSystem.InputSystem.RemoveDevice(keyboard);
    input.GetType().GetField("hasFocus",flags).SetValue(input,oldFocus);
    input.GetType().GetField("paused",flags).SetValue(input,oldPaused);
    if(target!=null)UnityEngine.Object.DestroyImmediate(target);
    move(originalPosition);Physics2D.simulationMode=oldSimulation;
}
return string.Join("\n",results.ToArray());
