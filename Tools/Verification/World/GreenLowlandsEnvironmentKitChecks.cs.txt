// Edit Mode asset integration checks. No scene mutations or save access.
if(EditorApplication.isPlaying)throw new System.Exception("Edit Mode required.");
string root=RoadOfTheOldKing.EditorTools.GreenLowlandsEnvironmentKitBuilder.Root;
var catalog=RoadOfTheOldKing.EditorTools.GreenLowlandsEnvironmentKitBuilder.ReadCatalog();
string regionalPrefabs=RoadOfTheOldKing.EditorTools.GreenLowlandsEnvironmentKitBuilder.PrefabRoot;
var treeNames=new System.Collections.Generic.HashSet<string>(catalog.regionalTrees.Select(t=>t.art.name));
var failures=new System.Collections.Generic.List<string>();int count=0;
System.Action<bool,string> check=(ok,label)=>{count++;if(!ok)failures.Add(label);};
check(catalog.entries.Length==9&&catalog.reusedPrefabs.Length==14,"Nine regional / 14 reused objects");
check(!System.IO.File.Exists(root+"/GreenLowlandsEnvironment16.png"),"Retired regional atlas absent");
string palettePath=root+"/Palettes/GreenLowlandsEnvironment.prefab";
check(AssetDatabase.LoadAllAssetsAtPath(palettePath).OfType<GridPalette>().Any(),"Registered palette");
var paletteMap=AssetDatabase.LoadAssetAtPath<GameObject>(palettePath).GetComponentInChildren<UnityEngine.Tilemaps.Tilemap>();
var body=AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Player/Player.prefab").GetComponentsInChildren<SpriteRenderer>().First(r=>r.name=="Body");
int index=0;
foreach(var prop in catalog.entries){
var prefab=AssetDatabase.LoadAssetAtPath<GameObject>(regionalPrefabs+"/"+prop.name+".prefab");
check(prefab!=null&&prefab.transform.localScale==Vector3.one,"Native prefab "+prop.name);if(!prefab)continue;
var maps=prefab.GetComponentsInChildren<UnityEngine.Tilemaps.Tilemap>().ToDictionary(m=>m.name);
var bases=maps["Environment"];var upper=maps["Above Player"];var collision=maps["Collision"];
check(!bases.GetComponent<Collider2D>()&&!upper.GetComponent<Collider2D>(),"Visual maps have no collision "+prop.name);
check(!collision.GetComponent<UnityEngine.Tilemaps.TilemapRenderer>().enabled&&collision.GetComponent<UnityEngine.Tilemaps.TilemapCollider2D>()!=null&&collision.gameObject.layer==LayerMask.NameToLayer("Environment"),"Separate physical footprint "+prop.name);
var render=upper.GetComponent<UnityEngine.Tilemaps.TilemapRenderer>();
check(render.sortingLayerID==body.sortingLayerID&&render.sortingOrder>body.sortingOrder,"Overhead above actual player renderer "+prop.name);
check(prefab.GetComponentsInChildren<MonoBehaviour>(true).Length==0,"Art-only, no gameplay attached "+prop.name);
var origin=new Vector3Int(index%4*12,-index/4*9,0);
for(int y=0;y<prop.height/16;y++)for(int x=0;x<prop.width/16;x++){
var p=new Vector3Int(x,y,0);var tile=(UnityEngine.Tilemaps.Tile)(bases.GetTile(p)??upper.GetTile(p));
check(bases.HasTile(p)!=upper.HasTile(p),"No missing/duplicate visual cell "+prop.name+p);
check(tile!=null&&tile.sprite!=null&&tile.sprite.rect.width==16&&tile.sprite.rect.height==16&&tile.colliderType==UnityEngine.Tilemaps.Tile.ColliderType.None&&paletteMap.GetTile(origin+p)==tile,"Palette/prefab piece agrees "+prop.name+p);
bool blocked=prop.collision.Any(a=>x>=a.x&&x<a.x+a.width&&y>=a.y&&y<a.y+a.height);
check(collision.HasTile(p)==blocked,"Footprint "+prop.name+p);
}
if(prop.name=="Plank_Bridge"){
for(int x=0;x<6;x++)foreach(int y in new[]{1,2})check(!collision.HasTile(new Vector3Int(x,y,0)),"Two-cell deck and open ends");
check(bases.HasTile(new Vector3Int(2,3,0))&&upper.HasTile(new Vector3Int(2,0,0)),"Far rail below / near rail above player");
}
index++;
}
foreach(string name in catalog.reusedPrefabs){
var prefab=AssetDatabase.LoadAssetAtPath<GameObject>(RoadOfTheOldKing.EditorTools.EnvironmentArtCatalog.PrefabPath(name));
check(prefab!=null,"Retained prefab "+name);
var origin=new Vector3Int(index%4*12,-index/4*9,0);
foreach(string mapName in new[]{"Environment","Above Player"}){
var map=prefab.transform.Find(mapName).GetComponent<UnityEngine.Tilemaps.Tilemap>();
foreach(var p in map.cellBounds.allPositionsWithin)if(map.HasTile(p))check(paletteMap.GetTile(origin+p)==map.GetTile(p),"Reused original tile reference "+name+p);
}
index++;
}
check(AssetDatabase.FindAssets("t:Prefab",new[]{regionalPrefabs}).Length==9,"Nine regional prefabs");
check(treeNames.SetEquals(new[]{"Oak","Birch","Apple_Tree"}),"Three shared tree families");
check(AssetDatabase.FindAssets("t:Scene",new[]{root}).Length==0,"No per-layer demo scene");
foreach(string path in catalog.entries.Select(p=>regionalPrefabs+"/"+p.name+".prefab").Concat(treeNames.Select(n=>RoadOfTheOldKing.EditorTools.EnvironmentArtCatalog.PrefabPath(n)))){
var prefab=AssetDatabase.LoadAssetAtPath<GameObject>(path);
check(prefab!=null,"Canonical prefab "+path);if(!prefab)continue;
check(PrefabUtility.GetPrefabAssetType(prefab)==PrefabAssetType.Regular&&PrefabUtility.GetCorrespondingObjectFromSource(prefab)==null,"Independent prefab "+prefab.name);
var art=prefab.GetComponentsInChildren<UnityEngine.Tilemaps.Tilemap>(true).Where(m=>m.name=="Environment"||m.name=="Above Player");
check(art.All(m=>m.GetTilesBlock(m.cellBounds).Where(t=>t!=null).All(t=>AssetDatabase.GetAssetPath(t).StartsWith(treeNames.Contains(prefab.name)?RoadOfTheOldKing.EditorTools.GreenLowlandsEnvironmentKitBuilder.TreeRoot+"/Tiles/":"Assets/Art/Tiles/GreenLowlands/Environment/Architecture/Tiles/"))),"Regional architecture or shared tree art only "+prefab.name);
}
return (failures.Count==0?"PASS ":"FAILED ")+count+" Environment assertions; 23 stamps / nine regional and three shared tree prefabs\n"+string.Join("\n",failures.ToArray());
