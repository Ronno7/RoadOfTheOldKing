    var changed=new System.Collections.Generic.List<string>();
    for(int i=0;i<SessionState.GetInt(p+"ProtectedCount",0);i++){
        string saved=SessionState.GetString(p+"Protected"+i,"");bool exists=SessionState.GetBool(p+"Exists"+i,false);string value=SessionState.GetString(p+"Value"+i,"");
        if(PlayerPrefs.HasKey(saved)!=exists||(exists&&PlayerPrefs.GetString(saved)!=value)){changed.Add(saved);if(exists)PlayerPrefs.SetString(saved,value);else PlayerPrefs.DeleteKey(saved);}
    }
    PlayerPrefs.DeleteKey(key);PlayerPrefs.DeleteKey(key+".Resume");PlayerPrefs.Save();
    if(SessionState.GetString(p+"Bytes","")!=Convert.ToBase64String(System.Security.Cryptography.SHA256.Create().ComputeHash(System.IO.File.ReadAllBytes(path))))changed.Add("SCENE GreenLowlands");

    UnityEditor.SceneManagement.EditorSceneManager.NewScene(UnityEditor.SceneManagement.NewSceneSetup.EmptyScene,UnityEditor.SceneManagement.NewSceneMode.Single);
    var setup=new UnityEditor.SceneManagement.SceneSetup[SessionState.GetInt(p+"Scenes",0)];
    for(int i=0;i<setup.Length;i++)setup[i]=new UnityEditor.SceneManagement.SceneSetup{path=SessionState.GetString(p+"Scene"+i,""),isLoaded=SessionState.GetBool(p+"Loaded"+i,false),isActive=SessionState.GetBool(p+"Selected"+i,false)};
    UnityEditor.SceneManagement.EditorSceneManager.RestoreSceneManagerSetup(setup);
    UnityEditor.SceneManagement.EditorSceneManager.playModeStartScene=AssetDatabase.LoadAssetAtPath<SceneAsset>(SessionState.GetString(p+"PlayStart",""));
    Application.runInBackground=SessionState.GetBool(p+"Background",false);
    SessionState.SetBool(p+"Active",false);SessionState.EraseString(p+"Command");
    return new{passed=SessionState.GetInt(p+"Passed",0),protectedEntriesChanged=changed.ToArray(),testKeysRemoved=!PlayerPrefs.HasKey(key)&&!PlayerPrefs.HasKey(key+".Resume"),restored=UnityEngine.SceneManagement.SceneManager.GetActiveScene().path,dirty=UnityEngine.SceneManagement.SceneManager.GetActiveScene().isDirty};
}

var beam=UnityEngine.Object.FindFirstObjectByType<RoadOfTheOldKing.World.RecallRelease>();

var body=player.GetComponent<Rigidbody2D>();

var axeTick=axe.GetType().GetMethod("Simulate",flags);

var view=beam.GetComponent<RoadOfTheOldKing.World.RecallReleaseView>();

var visualFrames=new System.Collections.Generic.HashSet<int>();bool movingCollisionClosed=true;

var log=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,name)=>{if(!ok)throw new System.Exception("FAILED: "+name+" | "+string.Join("; ",log.ToArray()));log.Add(name);SessionState.SetInt(p+"Passed",SessionState.GetInt(p+"Passed",0)+1);};
System.Action<Vector2> pose=pos=>{player.transform.position=pos;body.position=pos;body.linearVelocity=Vector2.zero;combat.GetType().GetField("controlsActive",flags).SetValue(combat,true);Physics2D.SyncTransforms();};

var filter=new ContactFilter2D{useTriggers=false};var sweep=new System.Collections.Generic.List<RaycastHit2D>();
System.Func<Vector2,Vector2,bool> blocked=(a,b)=>{Physics2D.SyncTransforms();Physics2D.BoxCast(a,new Vector2(.76f,.76f),0,(b-a).normalized,filter,sweep,Vector2.Distance(a,b));return sweep.Any(h=>h.collider!=null&&!h.collider.isTrigger&&h.collider.GetComponentInParent<RoadOfTheOldKing.Player.PlayerHealth>()==null);};
Vector2 south=new Vector2(111f,12.5f), northWest=new Vector2(109.5f,17.5f), receiver=new Vector2(116.5f,14.5f);

if(phase==0){
    SessionState.SetInt(p+"Passed",0);
    game.Progress.completedIds.RemoveAll(id=>id.StartsWith("green-lowlands/hayfields/"));
    beam.RestoreProgress(game.Progress);gate.RestoreProgress(game.Progress);
    check(!beam.IsReleased&&!gate.IsOpen,"fresh beam raised and gate barred");

if(phase==2){
    var motor=player.GetComponent<RoadOfTheOldKing.Player.PlayerMovement>();
    var input=player.GetComponent<RoadOfTheOldKing.Input.PlayerMovementInput>();

    var oldFreeVelocity=(Vector2)motor.GetType().GetField("freeVelocity",flags).GetValue(motor);
    var sluice=UnityEngine.Object.FindFirstObjectByType<RoadOfTheOldKing.World.RecallSluice>();

    };

                keys(held.ToArray());step();
            }
            if(frames>=400)throw new System.Exception("Motor blocked: "+label+" at "+body.position+" toward "+points[n]+" touching "+string.Join(", ",System.Linq.Enumerable.ToArray(System.Linq.Enumerable.Select(System.Linq.Enumerable.Where(Physics2D.OverlapCircleAll(body.position,.8f),c=>!c.isTrigger&&c.attachedRigidbody!=body),c=>c.name+" "+c.bounds.min.ToString("F1")+"-"+c.bounds.max.ToString("F1")))));
        }
        keys(new UnityEngine.InputSystem.Key[0]);check(true,label);
    };
    try{
        settings.editorInputBehaviorInPlayMode=UnityEngine.InputSystem.InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;
        actions.devices=new UnityEngine.InputSystem.InputDevice[]{keyboard};input.GetType().GetField("paused",flags).SetValue(input,false);Physics2D.simulationMode=SimulationMode2D.Script;

            walk(new[]{new Vector2(115.5f,13.6f),new Vector2(121,13.6f)},water+": motor crosses the lowered deck east");
            walk(new[]{new Vector2(121,13.6f),new Vector2(115.5f,13.6f)},water+": motor crosses the lowered deck west");

    }
    SessionState.SetInt(p+"Phase",3);
    return new{checks=log.ToArray(),motorSteps=steps,next="Stop; set command cleanup; run fixture."};
