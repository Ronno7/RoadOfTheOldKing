// Edit Mode only; all temporary objects live in a preview scene. Does not touch saves or gameplay.
// SessionState TutorialTerrainKit.Scope: current = shallow earth, stone = northern shelf, legacy = original kit.
if (SessionState.GetString("TutorialTerrainKit.Scope", "legacy") != "legacy")
{
if(EditorApplication.isPlayingOrWillChangePlaymode)throw new Exception("Stopped Edit Mode required.");
string root="Assets/Art/Tiles/Tutorial/Current";
bool stone=SessionState.GetString("TutorialTerrainKit.Scope", "current")=="stone";
string family=stone?"RaisedStone":"ShallowEarth";int front=stone?3:1;
var brush=AssetDatabase.LoadAssetAtPath<RoadOfTheOldKing.World.RaisedCliffTile>(root+"/Rules/Paint_"+family+".asset");
var sprites=AssetDatabase.LoadAllAssetsAtPath(root+(stone?"/TutorialStoneCliffs16.png":"/TutorialCliffs16.png")).OfType<Sprite>().ToArray();
int passed=0;Action<bool,string> check=(ok,label)=>{if(!ok)throw new Exception(label);passed++;};
check(brush!=null&&sprites.Length==188,"Complete joined cliff asset");
check(sprites.All(s=>s.rect.width==16&&s.rect.height==16*(front+1)&&s.pixelsPerUnit==16),"188 native complete columns");
check(sprites.All(s=>s.pivot==new Vector2(8,front*16+8)&&Mathf.Abs(s.bounds.min.y+front+.5f)<.001f),"Top-cell anchors");
var importer=(TextureImporter)AssetImporter.GetAtPath(root+(stone?"/TutorialStoneCliffs16.png":"/TutorialCliffs16.png"));
check(importer.filterMode==FilterMode.Point&&!importer.mipmapEnabled&&importer.textureCompression==TextureImporterCompression.Uncompressed,"Native import settings");
var previous=UnityEngine.SceneManagement.SceneManager.GetActiveScene();bool dirty=previous.isDirty;
var scratch=UnityEditor.SceneManagement.EditorSceneManager.NewPreviewScene();
try{
 var go=new GameObject("Shallow cliff authoring check");UnityEngine.SceneManagement.SceneManager.MoveGameObjectToScene(go,scratch);go.AddComponent<Grid>();
 var child=new GameObject("Tiles");child.transform.SetParent(go.transform,false);
 var map=child.AddComponent<UnityEngine.Tilemaps.Tilemap>();
 var offsets=new[]{new Vector3Int(0,1),new Vector3Int(1,0),new Vector3Int(0,-1),new Vector3Int(-1,0),new Vector3Int(1,1),new Vector3Int(1,-1),new Vector3Int(-1,-1),new Vector3Int(-1,1)};
 var seen=new System.Collections.Generic.HashSet<Sprite>();bool populated=true,visualOnly=true;
 for(int phase=-2;phase<2;phase++)for(int mask=0;mask<256;mask++){
  map.ClearAllTiles();var c=new Vector3Int(phase,0,0);map.SetTile(c,brush);
  for(int n=0;n<8;n++)if((mask&(1<<n))!=0)map.SetTile(c+offsets[n],brush);
  var sprite=map.GetSprite(c);populated&=sprite!=null;if(sprite!=null)seen.Add(sprite);
  visualOnly&=map.GetColliderType(c)==UnityEngine.Tilemaps.Tile.ColliderType.None;
 }
 check(populated&&seen.Count==188,"All neighborhoods resolve all sprites");
 check(visualOnly,"No collision added");
 map.ClearAllTiles();foreach(var p in offsets)map.SetTile(p,brush);map.SetTile(Vector3Int.zero,brush);
 check(map.GetSprite(Vector3Int.zero).name==family+"_255_0","Interior selects full top");
 var before=offsets.Select(p=>map.GetSprite(p)).ToArray();map.SetTile(Vector3Int.zero,null);
 check(offsets.Where((p,i)=>map.GetSprite(p)!=before[i]).Count()==8,"Erasing refreshes eight corners");
 map.SetTile(Vector3Int.zero,brush);
 check(offsets.Select((p,i)=>map.GetSprite(p)==before[i]).All(v=>v),"Repainting restores joins");
 map.ClearAllTiles();map.SetTile(new Vector3Int(-1,0),brush);
 check(map.GetSprite(new Vector3Int(-1,0)).name==family+"_000_3","Negative-column phase");
 map.SetTile(Vector3Int.zero,brush);
 check(map.GetSprite(new Vector3Int(-1,0)).name==family+"_002_3"&&map.GetSprite(Vector3Int.zero).name==family+"_008_0","Horizontal material continuity");
 var shed=AssetDatabase.LoadAllAssetsAtPath(root+"/TutorialWoodshed16.png").OfType<Sprite>().ToArray();
 check(shed.Length==12&&shed.All(s=>s.rect.size==new Vector2(16,16)&&s.pixelsPerUnit==16),"Woodshed twelve native slices");
}
finally{UnityEditor.SceneManagement.EditorSceneManager.ClosePreviewScene(scratch);}
check(previous.isDirty==dirty&&UnityEngine.SceneManagement.SceneManager.GetActiveScene()==previous,"Production scene preserved");
return new{passed,neighborhoods=1024,family,front,sprites=sprites.Length,scene=previous.path,dirty=previous.isDirty};
}
else
{
if(EditorApplication.isPlaying)throw new System.Exception("Edit Mode required.");
string root="Assets/Art/Tiles/Tutorial/Terrain";
var results=new System.Collections.Generic.List<string>();int assertions=0;
System.Action<bool,string> check=(ok,name)=>{assertions++;if(!ok)results.Add("FAIL "+name);};
var sprites=AssetDatabase.LoadAllAssetsAtPath(root+"/TutorialTerrain16.png").OfType<Sprite>().ToArray();
check(sprites.Length==315,"315 native sprites");
foreach(var s in sprites){check(s.rect.width==16&&s.rect.height==16&&s.pixelsPerUnit==16,"Native resolution "+s.name);check(s.bounds.size.x==1&&s.bounds.size.y==1,"Full-rect sprite "+s.name);}
var importer=(TextureImporter)AssetImporter.GetAtPath(root+"/TutorialTerrain16.png");
check(importer.filterMode==FilterMode.Point&&!importer.mipmapEnabled&&importer.textureCompression==TextureImporterCompression.Uncompressed&&importer.npotScale==TextureImporterNPOTScale.None,"Pixel import settings");
var tileIds=AssetDatabase.FindAssets("t:Tile",new[]{root+"/Tiles"});
check(tileIds.Length==315,"315 visual Tile assets");
foreach(var id in tileIds){var t=AssetDatabase.LoadAssetAtPath<UnityEngine.Tilemaps.Tile>(AssetDatabase.GUIDToAssetPath(id));check(t.sprite!=null&&t.colliderType==UnityEngine.Tilemaps.Tile.ColliderType.None,"Non-colliding art "+t.name);}
foreach(string name in new[]{"TutorialTerrain_Paint","TutorialTerrain_Structures","TutorialTerrain_IndividualPieces"})check(AssetDatabase.LoadAllAssetsAtPath(root+"/Palettes/"+name+".prefab").OfType<GridPalette>().Any(),"Registered palette "+name);
var template=AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Art/Tiles/Tutorial/Templates/TutorialZoneTemplate.prefab");
check(template!=null&&template.GetComponent<Grid>()!=null,"Combined ground and terrain template");
foreach(string category in new[]{"Ground","Terrain","Paths","Environment","Detail Decoration","Collision","Above Player","Interactive Objects"})check(template.transform.Find(category)!=null,"Category "+category);
check(template.transform.Find("Ground").GetComponentsInChildren<UnityEngine.Tilemaps.Tilemap>().Length==7,"Existing seven ground maps");
check(template.transform.Find("Terrain").GetComponentsInChildren<UnityEngine.Tilemaps.Tilemap>().Length==9,"Nine independent terrain maps");
check(template.transform.Find("Interactive Objects").GetComponent<UnityEngine.Tilemaps.Tilemap>()==null,"Interactive objects container");
check(template.GetComponentsInChildren<UnityEngine.Tilemaps.TilemapCollider2D>().Length==1&&!template.transform.Find("Collision").GetComponent<UnityEngine.Tilemaps.TilemapRenderer>().enabled,"Single hidden collision map");
var terrain=template.transform.Find("Terrain");
check(terrain.Find("MeadowWater").GetComponent<UnityEngine.Tilemaps.TilemapRenderer>().sortingOrder>terrain.Find("GrassLedge").GetComponent<UnityEngine.Tilemaps.TilemapRenderer>().sortingOrder,"Water can render on plateaus");
check(terrain.Find("WaterDetails").GetComponent<UnityEngine.Tilemaps.Tilemap>().animationFrameRate==4,"Gentle 4fps water animation");
foreach(string kind in new[]{"Ripple","FallBody","FallFoam"}){
var tile=AssetDatabase.LoadAssetAtPath<UnityEngine.Tilemaps.AnimatedTile>(root+"/Animations/Animated_"+kind+".asset");
check(tile!=null&&tile.m_AnimatedSprites.Length==4&&tile.m_AnimatedSprites.All(s=>s!=null),"Animation frames "+kind);
check(tile.m_TileColliderType==UnityEngine.Tilemaps.Tile.ColliderType.None,"No animated collision "+kind);
}
var scene=UnityEditor.SceneManagement.EditorSceneManager.NewPreviewScene();
try{
var grid=new GameObject("Terrain verification",typeof(Grid));UnityEngine.SceneManagement.SceneManager.MoveGameObjectToScene(grid,scene);
var go=new GameObject("Tiles",typeof(UnityEngine.Tilemaps.Tilemap),typeof(UnityEngine.Tilemaps.TilemapRenderer));go.transform.SetParent(grid.transform,false);
var map=go.GetComponent<UnityEngine.Tilemaps.Tilemap>();
Vector3Int[] offsets={new Vector3Int(0,1,0),new Vector3Int(1,0,0),new Vector3Int(0,-1,0),new Vector3Int(-1,0,0),new Vector3Int(1,1,0),new Vector3Int(1,-1,0),new Vector3Int(-1,-1,0),new Vector3Int(-1,1,0)};
foreach(string material in new[]{"MeadowWater","StoneWater","DeepWater","GrassLedge","StoneLedge","StoneWall"}){
var rule=AssetDatabase.LoadAssetAtPath<RuleTile>(root+"/Rules/Paint_"+material+".asset");
check(rule!=null&&rule.m_TilingRules.Count==(material=="StoneWall"?16:47),"Complete rule coverage "+material);
check(rule.m_TilingRules.Select(r=>r.m_Id).Distinct().Count()==rule.m_TilingRules.Count,"Unique rule IDs "+material);
check(rule.m_DefaultColliderType==UnityEngine.Tilemaps.Tile.ColliderType.None&&rule.m_TilingRules.All(r=>r.m_ColliderType==UnityEngine.Tilemaps.Tile.ColliderType.None),"Non-colliding brush "+material);
for(int raw=0;raw<256;raw++){
map.ClearAllTiles();map.SetTile(Vector3Int.zero,rule);
for(int bit=0;bit<8;bit++)if((raw&(1<<bit))!=0)map.SetTile(offsets[bit],rule);
int normalized=raw;
if(material=="StoneWall")normalized&=15;
else for(int bit=4;bit<8;bit++)if((raw&(1<<(bit-4)))==0||(raw&(1<<((bit-3)%4)))==0)normalized&=~(1<<bit);
map.RefreshAllTiles();var sprite=map.GetSprite(Vector3Int.zero);
check(sprite!=null&&(normalized==255?sprite.name.StartsWith(material+"_Center"):sprite.name.EndsWith("_"+normalized.ToString("D3"))),material+" neighbors "+raw);
}
map.ClearAllTiles();map.SetTile(Vector3Int.zero,rule);map.SetTile(Vector3Int.right,rule);var joined=map.GetSprite(Vector3Int.zero);map.SetTile(Vector3Int.right,null);var single=map.GetSprite(Vector3Int.zero);
check(single!=joined&&single.name.EndsWith("_000"),"Neighbor erase refresh "+material);
}
}finally{UnityEditor.SceneManagement.EditorSceneManager.ClosePreviewScene(scene);}
results.Insert(0,(results.Count==0?"PASS ":"FAILED ")+assertions+" terrain kit assertions; 1536 neighbor arrangements exercised.");
return string.Join("\n",results.ToArray());
}
