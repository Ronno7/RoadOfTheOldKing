 var body=player.GetComponent<Rigidbody2D>();

 System.Func<int,GameObject> visual=i=>(GameObject)typeof(RoadOfTheOldKing.World.CoinSource).GetField("coinVisual",flags).GetValue(sources[i]);
 var sweepType=typeof(RoadOfTheOldKing.Weapons.WeaponSweep);

 var raise=sweepType.GetMethod("Raise",System.Reflection.BindingFlags.Static|System.Reflection.BindingFlags.NonPublic);
 System.Action<Vector2,GameObject,bool> cut=(p,owner,clear)=>raise.Invoke(null,new[]{ctor.Invoke(new object[]{owner,p,p,Vector2.up,.01f,360f,.01f,false,new System.Func<Vector2,bool>(q=>clear)})});
 if(phase==2){
  check(patches.Length==16&&sources.Select(s=>s.SourceId).Distinct().Count()==16,"sixteen unique patch sources");

   var p=(Vector2)sources[i].transform.position;
   check(roots.Count(r=>(r-p).sqrMagnitude<=1.5625f)>=3,"patch "+i+" contains several rooted plants");
   move(p);check(!sources[i].TryCollect(player),"walking on uncut patch "+i+" cannot roll or collect");

  move(new Vector2(15.5f,3));
  cut(sources[0].transform.position,field.gameObject,true);
  check(game.Progress.coinSources.Count==0,"non-player sweep cannot earn grass coins");
  cut(sources[0].transform.position,player.gameObject,false);
  check(game.Progress.coinSources.Count==0,"terrain-blocked cut cannot roll");
  var rng=UnityEngine.Random.state;
  try{
   for(int i=0;i<3;i++){
    int seed=0;for(;seed<10000;seed++){UnityEngine.Random.InitState(seed);float r=UnityEngine.Random.value;if(i==0?r<.8f:i==1?r>=.8f&&r<.95f:r>=.95f)break;}
    UnityEngine.Random.InitState(seed);cut(sources[i].transform.position,player.gameObject,true);
    var record=game.Progress.FindCoinSource(sources[i].SourceId);int expected=i==0?0:i==1?1:5;
    check(record!=null&&record.amount==expected&&record.collected==(i==0),"saved roll tier "+expected);
    check(visual(i).activeSelf==(i!=0),"correct visible drop for tier "+expected);
    int n=game.Progress.coinSources.Count;

   }
  }finally{UnityEngine.Random.state=rng;}

  move(sources[1].transform.position);int wallet=game.Progress.bronzeCoins;

  bool controls=(bool)combat.GetType().GetField("controlsActive",flags).GetValue(combat);

   combat.GetType().GetField("controlsActive",flags).SetValue(combat,true);move((Vector2)sources[3].transform.position-Vector2.right*.7f);axe.CancelAction();player.GetComponent<RoadOfTheOldKing.Player.PlayerStamina>().Restore();

 }else {

  else{move(sources[2].transform.position);check(sources[2].TryCollect(player)&&game.Progress.bronzeCoins==SessionState.GetInt(prefix+"Wallet",-1)+5,"pending drop collected after Continue and death");check(!sources[2].TryCollect(player),"restored drop cannot duplicate");}
 }
}
SessionState.SetInt(prefix+"Phase",phase+1);int total=SessionState.GetInt(prefix+"Passed",0)+checks.Count;SessionState.SetInt(prefix+"Passed",total);
